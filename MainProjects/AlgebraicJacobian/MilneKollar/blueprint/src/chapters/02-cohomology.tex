\chapter{Sheaf cohomology and base change}
\label{mk-ch-cohomology}

For a \(k\)-scheme \(T\), put \(C_T=C\times_kT\).  Cohomology is taken on
the Zariski site (equivalently for quasi-coherent sheaves on the small fppf
site).

\begin{definition}[Module-valued sheaf cohomology]
  \label{mk-def-module-cohomology}
  \dcref{III.1,custom}
  \cite{stacks-cohomology}
  For a sheaf \(\mathcal F\) of \(k\)-modules on \(C\), define
  \(H^i(C,\mathcal F)=\operatorname{Ext}^i(k_C,\mathcal F)\), where \(k_C\)
  is the constant sheaf.
\end{definition}

\begin{proposition}[The Cech short exact sequence]
  \label{mk-prop-cech-short-exact}
  \dcref{III.1}
  \uses{mk-def-cech-complex,mk-lem-affine-h1-vanishing}
  \cite{stacks-cohomology-schemes}
  For an affine cover \(C=U\cup V\) and quasi-coherent \(\mathcal F\),
  \[
    0\to\Gamma(C,\mathcal F)\to\mathcal F(U)\oplus\mathcal F(V)
    \to\mathcal F(U\cap V)\to H^1(C,\mathcal F)\to0.
  \]
\end{proposition}

\begin{proof}
  The first three terms are the sheaf equalizer, and the last two are the
  kernel and cokernel of the Cech differential.  Affine vanishing makes the
  Cech complex compute derived cohomology.
\end{proof}

\begin{definition}[Two-affine Cech complex]
  \label{mk-def-cech-complex}
  \dcref{III.1}
  \cite{stacks-cohomology-schemes}
  For an affine cover \(C=U\cup V\) and a quasi-coherent sheaf \(\mathcal F\),
  set
  \[
    \check C^0(\mathcal F)=\mathcal F(U)\oplus\mathcal F(V),\qquad
    \check C^1(\mathcal F)=\mathcal F(U\cap V),
  \]
  with differential \(d(s,t)=s|_{U\cap V}-t|_{U\cap V}\).
\end{definition}

\begin{lemma}[Cech differential and degree zero]
  \label{mk-lem-cech-differential}
  \dcref{III.1}
  \uses{mk-def-cech-complex}
  \cite{stacks-cohomology-schemes}
  The two-affine complex has no term in degree \(2\), and
  \[
    \check H^0(\mathcal F)=
    \ker\bigl(\mathcal F(U)\oplus\mathcal F(V)\to\mathcal F(U\cap V)\bigr).
  \]
\end{lemma}

\begin{proof}
  There is no degree-two term.  The kernel condition is exactly equality of
  the two restrictions, which is the sheaf gluing condition.
\end{proof}

\begin{proposition}[Two-affine Cech computes cohomology]
  \label{mk-prop-cech-computes}
  \dcref{III.1}
  \uses{mk-def-cech-complex,mk-lem-cech-differential}
  \cite{stacks-cohomology-schemes}
  If \(U,V,U\cap V\) are affine, then
  \[
    H^0(C,\mathcal F)=\check H^0(\mathcal F),\qquad
    H^1(C,\mathcal F)=\operatorname{coker}(d).
  \]
\end{proposition}

\begin{proof}
  A quasi-coherent sheaf is acyclic on each affine in the cover and on their
  intersection.  The Cech-to-derived spectral sequence therefore degenerates
  to the cohomology of the displayed two-term complex.
\end{proof}

\begin{lemma}[Affine vanishing in degree one]
  \label{mk-lem-affine-h1-vanishing}
  \dcref{III.1}
  \cite{stacks-cohomology-schemes}
  For an affine scheme \(W\) and a quasi-coherent sheaf \(\mathcal F\),
  \(H^i(W,\mathcal F)=0\) for every \(i>0\).
\end{lemma}

\begin{proof}
  Write \(W=\Spec A\) and \(\mathcal F=\widetilde M\).  The section functor is
  \(M\mapsto M\), hence exact on the category of \(A\)-modules; its derived
  functors vanish in positive degree.
\end{proof}

\begin{lemma}[Relative curves have no higher coherent cohomology]
  \label{mk-lem-curve-h2-vanishing}
  \dcref{III.1}
  \uses{mk-prop-cech-computes}
  \cite{stacks-cohomology-schemes}
  For every \(k\)-scheme \(T\) and quasi-coherent \(\mathcal F\) on \(C_T\),
  \(H^i(C_T,\mathcal F)=0\) for \(i\ge2\).
\end{lemma}

\begin{proof}
  Base change preserves the finite affine cover and affine intersections.
  The resulting Cech complex has no terms above degree one, so its cohomology
  vanishes in degrees \(i\ge2\).
\end{proof}

\begin{proposition}[Flat scalar extension of the Cech complex]
  \label{mk-prop-cech-flat-base-change}
  \dcref{III.12}
  \uses{mk-def-cech-complex}
  \cite{stacks-cohomology-schemes}
  For a field extension \(k\subset k'\), the Cech complex for
  \(\mathcal F_{k'}\) is canonically
  \(\check C^\bullet(\mathcal F)\otimes_k k'\).
\end{proposition}

\begin{proof}
  Sections of a quasi-coherent sheaf on an affine are the corresponding
  module, and localization and tensor product commute for the flat extension
  \(k\to k'\).  This identifies both terms and the restriction differential.
\end{proof}

\begin{lemma}[Cohomology commutes with flat extension]
  \label{mk-lem-cech-flat-cohomology}
  \dcref{III.12}
  \uses{mk-prop-cech-flat-base-change,mk-prop-cech-computes}
  \cite{stacks-cohomology-schemes}
  Tensoring the two-affine Cech complex with \(k'\) gives
  \(H^i(C,\mathcal F)\otimes_k k'\cong H^i(C_{k'},\mathcal F_{k'})\).
\end{lemma}

\begin{proof}
  A field extension is flat, so kernels and cokernels of the Cech differential
  commute with tensor product.  Use \ref{mk-prop-cech-computes} on both curves.
\end{proof}

\begin{theorem}[Cohomology and base change]
  \label{mk-thm-cohomology-base-change}
  \dcref{III.12}
  \uses{mk-def-genus}
  \cite{stacks-project,stacks-cohomology-schemes}
  For a coherent sheaf \(\mathcal F\) on \(C\), \(H^i(C,\mathcal F)\) is
  finite-dimensional over \(k\).  If \(k\to k'\) is a field extension, then
  \[
    H^i(C,\mathcal F)\otimes_k k'\cong
    H^i(C_{k'},\mathcal F_{k'}).
  \]
\end{theorem}

\begin{proof}
  A finite affine cover of the proper curve gives a bounded Cech complex of
  finitely generated \(k\)-modules computing the cohomology of a coherent
  sheaf.  Hence each cohomology group is finite-dimensional.  After extending
  scalars to \(k'\), the same cover computes the base-changed sheaf and every
  term of the complex is tensored with \(k'\).  Since field extensions are
  flat, tensoring commutes with taking cohomology, which gives the canonical
  isomorphism.
\end{proof}

\begin{proposition}[Global functions]
  \label{mk-prop-global-functions}
  \dcref{III.1}
  \cite{hartshorne-ag,stacks-schemes}
  If \(C\) is geometrically integral and proper over \(k\), then
  \(H^0(C,\mathcal O_C)=k\).
\end{proposition}

\begin{proof}
  Properness makes \(H^0(C,\mathcal O_C)\) a finite \(k\)-algebra.  Apply the
  base-change theorem to \(i=0\) and an algebraic closure \(\bar k\):
  \[
    H^0(C,\mathcal O_C)\otimes_k\bar k
      \cong H^0(C_{\bar k},\mathcal O_{C_{\bar k}}).
  \]
  The scheme \(C_{\bar k}\) is proper and integral, so its global functions
  form a finite field extension of \(\bar k\), hence equal \(\bar k\).
  Consequently \(H^0(C,\mathcal O_C)\) has \(k\)-dimension one and the
  structural inclusion \(k\hookrightarrow H^0(C,\mathcal O_C)\) is an
  isomorphism.
\end{proof}

\begin{proposition}[The structure sheaf controls the tangent dimension]
  \label{mk-prop-genus-tangent}
  \dcref{5.11,custom}
  \uses{mk-def-genus,mk-thm-cohomology-base-change}
  \cite{kleiman-picard}
  For the identity of the Picard scheme, the Zariski tangent space satisfies
  \[
    T_0\operatorname{Pic}_{C/k}\cong H^1(C,\mathcal O_C).
  \]
  In particular its dimension is \(g(C)\), and this identification commutes
  with extension of the ground field.
\end{proposition}

\begin{proof}
  Let \(k[\epsilon]=k[\epsilon]/(\epsilon^2)\).  The kernel of
  \(\operatorname{Pic}_{C/k}(k[\epsilon])\to\operatorname{Pic}_{C/k}(k)\)
  consists of line bundles on \(C_{k[\epsilon]}\) restricting to the trivial
  bundle.  The exact sequence of sheaves
  \[
    1\longrightarrow 1+\epsilon\mathcal O_C
    \longrightarrow\mathcal O_{C_{k[\epsilon]}}^\times
    \longrightarrow\mathcal O_C^\times\longrightarrow1
  \]
  identifies \(1+\epsilon\mathcal O_C\) with the additive sheaf
  \(\mathcal O_C\).  Taking cohomology identifies this kernel, and hence the
  tangent space at the identity, with \(H^1(C,\mathcal O_C)\).  Replacing
  \(\epsilon\) by \(a\epsilon\) gives the \(k\)-vector-space structure, and
  the base-change statement follows from the preceding theorem.
\end{proof}
