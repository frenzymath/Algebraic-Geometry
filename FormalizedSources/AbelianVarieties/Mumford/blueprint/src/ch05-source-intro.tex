\section{Appendix I: The theorem of Tate}
\label{mumford-appendix-I-intro-source}
\dcref{M:appI}

\begin{definition}[Field of definition]
\label{mumford-def-field-of-definition}
\dcref{M:appI:I}
A field of definition of an abelian variety $X/k$ is a subfield $k_0$ over
which a group scheme $X_0$ exists with $X_0\otimes_{k_0}k\simeq X$.
\source{mumford-abelian-varieties:page-0251}
\end{definition}

\begin{proposition}[Tate-module invariants and field of definition]
\label{mumford-prop-tate-module-field-definition}
\dcref{M:appI:I}
For abelian varieties $X,Y$ over algebraically closed $k$ and
$\ell\ne{\rm char}(k)$,
\[
\mathbb Z_\ell\otimes_{\mathbb Z}{\rm Hom}(X,Y)\xrightarrow{T_\ell}
{\rm Hom}_{\mathbb Z_\ell}(T_\ell(X),T_\ell(Y)) \tag{1}
\]
is injective. Choose a common field of definition $k_0$, of finite type over
the prime field, and let $\bar k_0$ be its algebraic closure in $k$. Then
$G={\rm Gal}(\bar k_0/k_0)$ acts continuously on $T_\ell(X),T_\ell(Y)$.
Every homomorphism $X\to Y$ is defined over a finite extension $k_1/k_0$ in
$\bar k_0$; if $H={\rm Gal}(\bar k_0/k_1)$, then
\[
 h\varphi(x)=\varphi(hx)
\]
for every $h\in H$ and every $\bar k_0$-point $x$ of $X$.
\source{mumford-abelian-varieties:page-0251}
\uses{mumford-def-tate-module}
\end{proposition}
\begin{proof}
The kernels $X_n,Y_n$ of multiplication by $n$ are defined over $k_0$ and
their points are $\bar k_0$-rational. Compatibility of multiplication maps
gives the continuous Galois actions on the Tate modules. Finite-order points
are $k_0$-rational and dense; a homomorphism maps them to finite-order points,
so it is defined over $\bar k_0$, hence over some finite $k_1/k_0$. The final
equivariance identity follows from descent to $k_1$.

\end{proof}
