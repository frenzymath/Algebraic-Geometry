\chapter{Curves, divisors, and Riemann--Roch}
\label{pa-ch-curves}

Let \(k\) be a field and \(C\to\Spec k\) a smooth, proper, geometrically
integral curve.

\begin{definition}[Divisor line bundle]
  \label{pa-def-divisor-line-bundle}
  \dcref{II.6}
  \cite{hartshorne-ag}
  For \(D=\sum n_P[P]\), put
  \[
    \mathcal O_C(D)_P=\{f\in K(C):\operatorname{ord}_P(f)+n_P\ge0\}.
  \]
\end{definition}

\begin{lemma}[Divisor bundles and tensor products]
  \label{pa-lem-divisor-bundle-tensor}
  \dcref{II.6}
  \uses{pa-def-divisor-line-bundle}
  \cite{hartshorne-ag}
  The sheaf \(\mathcal O_C(D)\) is invertible and
  \(\mathcal O_C(D+E)\cong\mathcal O_C(D)\otimes\mathcal O_C(E)\).
\end{lemma}

\begin{proof}
  A uniformizer at \(P\) identifies the stalk with
  \(\pi_P^{-n_P}\mathcal O_{C,P}\); multiplication in \(K(C)\) gives the
  tensor identity.
\end{proof}

\begin{lemma}[Principal divisors have degree zero]
  \label{pa-lem-principal-degree-zero}
  \dcref{II.6}
  \uses{pa-def-divisors}
  \cite{hartshorne-ag,papaioannou-rr}
  \(\deg\operatorname{div}(f)=0\) for \(f\in K(C)^\times\).
\end{lemma}

\begin{proof}
  The product formula for the finite extension \(K(C)/k(t)\) gives the
  weighted sum of valuations as zero, which is precisely the displayed degree.
\end{proof}

\begin{lemma}[Adding a closed point]
  \label{pa-lem-divisor-exact-sequence}
  \dcref{II.6}
  \uses{pa-def-divisor-line-bundle}
  \cite{hartshorne-ag}
  For a closed point \(P\),
  \[
    0\to\mathcal O_C(D-P)\to\mathcal O_C(D)\to\kappa(P)\to0
  \]
  is exact.
\end{lemma}

\begin{proof}
  The first map is an isomorphism off \(P\).  At \(P\), the quotient of the
  two free rank-one stalks is
  \(\mathcal O_{C,P}/(\pi_P)\cong\kappa(P)\).
\end{proof}

\begin{definition}[Genus]
  \label{pa-def-genus}
  \dcref{IV.1,custom}
  \cite{milne-av,kleiman-picard}
  The genus of \(C\) is
  \[
    g(C)=\dim_k H^1(C,\mathcal O_C).
  \]
\end{definition}

Proper coherent-cohomology finiteness makes the displayed dimension finite;
geometric integrality identifies it with the usual genus after any field
extension.

\begin{definition}[Divisors and degree]
  \label{pa-def-divisors}
  \dcref{II.6}
  \cite{hartshorne-ag,papaioannou-rr}
  A divisor is \(D=\sum_xn_x[x]\) with finite support on the closed points of
  \(C\), and
  \[
    \deg D=\sum_xn_x[\kappa(x):k].
  \]
  Principal divisors are \(\operatorname{div}(f)\) for
  \(f\in K(C)^\times\); divisor classes form \(\operatorname{Pic}(C)\).
\end{definition}

\begin{theorem}[Riemann--Roch]
  \label{pa-thm-riemann-roch}
  \dcref{IV.1}
  \uses{pa-def-genus,pa-def-divisors}
  \cite{hartshorne-ag,papaioannou-rr}
  There is a canonical divisor \(K_C\) of degree \(2g(C)-2\) and
  \[
    \dim_kH^0(C,\mathcal O_C(D))-
    \dim_kH^0(C,\mathcal O_C(K_C-D))
    =\deg D+1-g(C).
  \]
\end{theorem}

\begin{proof}
  Serre duality gives
  \(H^1(C,\mathcal O_C(D))^\vee\simeq
  H^0(C,\mathcal O_C(K_C-D))\).  For a closed point \(P\), the exact sequence
  \[
    0\longrightarrow\mathcal O_C(D-P)\longrightarrow\mathcal O_C(D)
    \longrightarrow\kappa(P)\longrightarrow0
  \]
  changes the Euler characteristic by \([\kappa(P):k]\).  Serre vanishing
  starts the induction at large degree, giving
  \(\chi(\mathcal O_C(D))=\deg D+\chi(\mathcal O_C)\).  Since
  \(\chi(\mathcal O_C)=1-g(C)\), duality yields the formula.
\end{proof}
