\section{Primary ideals}
\label{am-ch4-primary}
\begin{definition}[Primary ideal]
\label{am-def-primary}
\source{atiyah-macdonald-commutative-algebra:page-0059}
An ideal $\q\ne A$ is primary if $xy\in\q$ and $x\notin\q$ imply
$y^n\in\q$ for some $n>0$.  Equivalently, every zero-divisor in $A/\q$ is
nilpotent.  A primary ideal is $\p$-primary when $\rad(\q)=\p$.
\end{definition}
\begin{proposition}[Radical of a primary ideal]
\label{am-prop-4-1}
\dcref{4.1}
\source{atiyah-macdonald-commutative-algebra:page-0059}
If $\q$ is primary, $\rad(\q)$ is the smallest prime ideal containing $\q$.
\end{proposition}
\begin{proof}
Put $\p=\rad(\q)$.  If $(xy)^m\in\q$, primaryness gives $x^m\in\q$ or
$y^n\in\q$ for some $n$, hence $x\in\p$ or $y\in\p$.  Thus $\p$ is prime;
it contains $\q$ and every prime containing $\q$ contains all radicals.
\end{proof}
\begin{proposition}[Maximal radical implies primary]
\label{am-prop-4-2}
\dcref{4.2}
\source{atiyah-macdonald-commutative-algebra:page-0060}
If $\rad(\aideal)$ is maximal, then $\aideal$ is primary.  In particular every
power of a maximal ideal is primary.
\end{proposition}
\begin{proof}
The image of $\rad(\aideal)$ in $A/\aideal$ is its nilradical.  By
Proposition~\ref{am-prop-1-8}, the quotient has only one prime ideal, so every
element is a unit or nilpotent; its zero-divisors are therefore nilpotent.
\uses{am-prop-1-8,am-def-primary}
\end{proof}
\begin{lemma}[Intersections of primary ideals]
\label{am-lem-4-3}
\dcref{4.3}
\source{atiyah-macdonald-commutative-algebra:page-0060}
The intersection of finitely many $\p$-primary ideals is $\p$-primary.
\end{lemma}
\begin{proof}
The radical of an intersection is the intersection of the radicals, hence $\p$;
if $xy$ lies in the intersection and $x$ is outside it, primaryness in one
component forces a power of $y$ into every component.
\end{proof}
\begin{lemma}[Colon ideals of a primary ideal]
\label{am-lem-4-4}
\dcref{4.4}
\source{atiyah-macdonald-commutative-algebra:page-0060}
If $\q$ is $\p$-primary, then $(\q:x)=(1)$ for $x\in\q$; if $x\notin\q$,
$(\q:x)$ is $\p$-primary; and if $x\notin\p$, $(\q:x)=\q$.
\end{lemma}
\begin{proof}
The first and third assertions are immediate.  If $x\notin\q$, then
$\q\subseteq(\q:x)\subseteq\p$ and radicals are equal.  If $yz\in(\q:x)$
with $y\notin\p$, then $xyz\in\q$ forces $xz\in\q$, so
$z\in(\q:x)$.
\end{proof}
\begin{definition}[Primary decomposition]
\label{am-def-primary-decomp}
\source{atiyah-macdonald-commutative-algebra:page-0060}
A primary decomposition of $\aideal$ is a finite intersection
$\aideal=\bigcap_{i=1}^n\q_i$ of primary ideals.  It is minimal when the
radicals $\p_i=\rad(\q_i)$ are distinct and no component can be omitted.
\end{definition}
\begin{theorem}[First uniqueness theorem]
\label{am-thm-4-5}
\dcref{4.5}
\source{atiyah-macdonald-commutative-algebra:page-0061}
If $\aideal=\bigcap_i\q_i$ is a minimal primary decomposition with
$\p_i=\rad(\q_i)$, then the $\p_i$ are exactly the prime ideals occurring as
$\rad(\aideal:x)$ for $x\in A$.  Hence they depend only on $\aideal$.
\end{theorem}
\begin{proof}
By Lemma~\ref{am-lem-4-4},
$(\aideal:x)=\bigcap_i(\q_i:x)$ and its radical is the intersection of those
$\p_i$ for which $x\notin\q_i$, with the other factors equal to $(1)$.
Prime avoidance (Proposition~\ref{am-prop-1-11}) shows every prime radical so
obtained is one of the $\p_i$.  For each $i$, choose
$x_i\in\bigcap_{j\ne i}\q_j\setminus\q_i$ by minimality; then
$\rad(\aideal:x_i)=\p_i$.
\uses{am-lem-4-4,am-prop-1-11}
\end{proof}
\begin{definition}[Associated, isolated, and embedded primes]
\label{am-def-associated-primes}
\source{atiyah-macdonald-commutative-algebra:page-0061}
The primes in Theorem~\ref{am-thm-4-5} are associated with $\aideal$.  The
minimal members are isolated (or minimal) primes; the others are embedded.
\end{definition}
\begin{proposition}[Minimal primes over a decomposable ideal]
\label{am-prop-4-6}
\dcref{4.6}
\source{atiyah-macdonald-commutative-algebra:page-0061}
Every prime containing a decomposable ideal $\aideal$ contains a minimal prime
associated with $\aideal$.  Thus the minimal associated primes are precisely
the minimal primes over $\aideal$.
\end{proposition}
\begin{proof}
If $\p\supseteq\aideal=\bigcap_i\q_i$, then
$\p\supseteq\bigcap_i\p_i$; Proposition~\ref{am-prop-1-11} gives
$\p\supseteq\p_i$ for some $i$, and descending among the finite set yields a
minimal one.
\uses{am-prop-1-11,am-def-primary-decomp}
\end{proof}
\begin{proposition}[Zero-divisors in a decomposable quotient]
\label{am-prop-4-7}
\dcref{4.7}
\source{atiyah-macdonald-commutative-algebra:page-0062}
For a decomposable ideal $\aideal$ with associated primes $\p_i$,
\[
\bigcup_i\p_i=\{x:(\aideal:x)\ne\aideal\}.
\]
In particular, if $(0)$ is decomposable, the zero-divisors are the union of
the associated primes and the nilradical is the intersection of the minimal
associated primes.
\end{proposition}
\begin{proof}
Apply Theorem~\ref{am-thm-4-5} to $A/\aideal$ and use the characterization of
zero-divisors as the union of radicals of annihilators from
Proposition~\ref{am-prop-1-14}.  The minimal-prime intersection is the
nilradical by Proposition~\ref{am-prop-1-8}.
\uses{am-thm-4-5,am-prop-1-14,am-prop-1-8}
\end{proof}
\begin{proposition}[Localization of primary ideals]
\label{am-prop-4-8}
\dcref{4.8}
\source{atiyah-macdonald-commutative-algebra:page-0062}
If $\q$ is $\p$-primary and $S$ is multiplicatively closed, then
$S^{-1}\q=S^{-1}A$ when $S\cap\p\ne\varnothing$.  If $S\cap\p=\varnothing$,
$S^{-1}\q$ is $S^{-1}\p$-primary and contracts to $\q$.  Thus localization
gives the corresponding bijection between primary and contracted primary
ideals.
\end{proposition}
\begin{proof}
If $s\in S\cap\p$, some power of $s$ lies in $\q$ and becomes a unit.  If
$S\cap\p=\varnothing$ and $sa\in\q$ with $s\in S$, primaryness gives
$a\in\q$, so contraction is $\q$; radicals localize by Proposition~\ref{am-prop-3-11},
and the defining zero-divisor condition is checked after clearing denominators.
\uses{am-prop-3-11}
\end{proof}
\begin{proposition}[Localized primary decompositions]
\label{am-prop-4-9}
\dcref{4.9}
\source{atiyah-macdonald-commutative-algebra:page-0063}
Let $\aideal=\bigcap_i\q_i$ be minimal, with $\p_i=\rad(\q_i)$, and suppose
$S$ meets exactly $\p_{m+1},\ldots,\p_n$.  Then
\[
S^{-1}\aideal=\bigcap_{i=1}^mS^{-1}\q_i
\]
is a minimal primary decomposition, and contraction recovers $\aideal$.
\end{proposition}
\begin{proof}
Localize the intersection using Proposition~\ref{am-prop-3-11}; the components
whose radicals meet $S$ become the unit ideal, and the others remain primary by
Proposition~\ref{am-prop-4-8}.  Distinct surviving radicals remain distinct.
\end{proof}
\begin{theorem}[Second uniqueness theorem]
\label{am-thm-4-10}
\dcref{4.10}
\source{atiyah-macdonald-commutative-algebra:page-0063}
For an isolated set of associated primes, the intersection of the corresponding
primary components is independent of the chosen minimal primary decomposition.
\end{theorem}
\begin{proof}
Let $S=A\setminus\bigcup_{\p\in\Sigma}\p$.  Isolation and prime avoidance
ensure precisely the primes in $\Sigma$ miss $S$.  Proposition~\ref{am-prop-4-9}
shows the desired intersection is $S(\aideal)$, which depends only on
$\aideal$ and $S$.
\uses{am-prop-4-9,am-prop-1-11}
\end{proof}
\begin{corollary}[Uniqueness of isolated components]
\label{am-cor-4-11}
\dcref{4.11}
\source{atiyah-macdonald-commutative-algebra:page-0063}
The primary components corresponding to minimal (isolated) primes are uniquely
determined by $\aideal$.
\uses{am-thm-4-10}
\end{corollary}
\begin{proof}
Apply Theorem~\ref{am-thm-4-10} to the set of minimal associated primes.
\end{proof}
