\section{The exterior algebra}

\begin{theorem}[Exterior powers of a direct sum]
\label{mcrt-thm-c-2}\dcref{C.2}
\source{matsumura-commutative-ring-theory:page-0299}\uses{}
For $A$-modules $M$ and $N$ and every $r\geq0$ there is a natural isomorphism
\[
\bigwedge_A^r(M\oplus N)
\cong\bigoplus_{s+t=r}\bigl(\bigwedge_A^sM\bigr)\otimes_A
\bigl(\bigwedge_A^tN\bigr).
\]
\end{theorem}
\begin{proof}
The alternating map obtained by wedging the $M$- and $N$-components induces
the displayed map by the universal property of exterior powers.  Its inverse
sends a pure tensor $(m_1\wedge\cdots\wedge m_s)\otimes
(n_1\wedge\cdots\wedge n_t)$ to
$m_1\wedge\cdots\wedge m_s\wedge n_1\wedge\cdots\wedge n_t$.
The defining alternating relations show that these maps are mutually inverse.
\end{proof}
