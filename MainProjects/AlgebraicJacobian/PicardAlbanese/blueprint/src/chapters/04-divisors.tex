\chapter{Divisors and Abel maps}
\label{pa-ch-divisors}

\begin{definition}[Relative effective divisor]
  \label{pa-def-relative-effective-divisor}
  \dcref{3.7,custom}
  \cite{kleiman-picard}
  For \(T\in(\Sch/k)\), a relative effective divisor on \(C_T\) is a closed
  subscheme finite and flat over \(T\).
\end{definition}

\begin{lemma}[Relative effective divisors are Cartier]
  \label{pa-lem-relative-divisor-cartier}
  \dcref{3.7,custom}
  \uses{pa-def-relative-effective-divisor,pa-lem-closed-point-dvr}
  \cite{kleiman-picard}
  Every relative effective divisor on the smooth relative curve \(C_T/T\) is
  an effective Cartier divisor.
\end{lemma}

\begin{proof}
  The ideal is finite flat over \(T\) and has fibrewise height one.  Smooth
  relative dimension one makes the local rings regular of dimension one, so
  each ideal is locally generated by a non-zero divisor.
\end{proof}

\begin{lemma}[The Hilbert functor of divisors]
  \label{pa-lem-divisor-hilbert-functor}
  \dcref{3.7,custom}
  \uses{pa-def-relative-effective-divisor,pa-lem-relative-divisor-cartier}
  \cite{kleiman-picard,nitsure-hilbert-quot}
  The degree-\(n\) relative-divisor functor is the length-\(n\) open-and-closed
  subfunctor of \(\operatorname{Hilb}_{C/k}\).
\end{lemma}

\begin{proof}
  A finite flat divisor has constant fibre length \(n\), hence a fixed Hilbert
  polynomial.  Conversely, a flat length-\(n\) family on a smooth curve is
  Cartier by \ref{pa-lem-relative-divisor-cartier}.
\end{proof}

\begin{lemma}[Symmetric powers represent divisors]
  \label{pa-lem-symmetric-power-represents}
  \dcref{III.6,custom}
  \uses{pa-def-symmetric-power,pa-lem-divisor-hilbert-functor}
  \cite{milne-av,kleiman-picard}
  The length-\(n\) Hilbert scheme of \(C\) is \(C^{(n)}\); it is smooth and
  proper of relative dimension \(n\).
\end{lemma}

\begin{proof}
  Locally on \(C\), a length-\(n\) subscheme is given by a monic degree-\(n\)
  polynomial, whose coefficients give smooth coordinates.  The Hilbert scheme
  is projective, and the symmetric-group quotient has the same functor of
  effective divisors.
\end{proof}

\begin{lemma}[Degree of the divisor line bundle]
  \label{pa-lem-effective-divisor-degree}
  \dcref{3.7,custom}
  \uses{pa-lem-relative-divisor-cartier,pa-def-divisor-line-bundle}
  \cite{kleiman-picard}
  If \(D\) has relative degree \(n\), then \(\mathcal O_{C_T}(D)\) has degree
  \(n\) on every geometric fibre.
\end{lemma}

\begin{proof}
  Restrict to a geometric fibre.  The Cartier divisor exact sequence identifies
  the length of \(\mathcal O_D\) with the degree of
  \(\mathcal O_C(D)\); flatness makes this length equal to \(n\).
\end{proof}

\begin{lemma}[Fibres of the effective divisor map]
  \label{pa-lem-effective-map-fibres}
  \dcref{3.7,custom}
  \uses{pa-prop-effective-map,pa-lem-principal-degree-zero}
  \cite{kleiman-picard}
  A geometric fibre of \(C^{(n)}\to\operatorname{Pic}^n_{C/k}\) is the complete
  linear system of any divisor in that fibre.
\end{lemma}

\begin{proof}
  Equality of classes is equivalent to differing by a principal divisor.
  Effective divisors linearly equivalent to \(D\) are precisely zero divisors of
  nonzero sections of \(\mathcal O_C(D)\), modulo scalar multiplication.
\end{proof}

\begin{lemma}[Base change of relative divisors]
  \label{pa-lem-relative-divisor-base-change}
  \dcref{3.7,custom}
  \uses{pa-def-relative-effective-divisor,pa-lem-relative-divisor-cartier}
  \cite{kleiman-picard}
  If \(T'\to T\), the pullback of a relative effective divisor
  \(D\subset C_T\) is a relative effective divisor on \(C_{T'}\), and
  \(\mathcal O(D)\) pulls back to \(\mathcal O(D_{T'})\).
\end{lemma}

\begin{proof}
  Finite flatness and the Cartier condition are stable under base change.
  Pullback of the invertible ideal of \(D\), followed by dualisation, gives
  the line-bundle identity.
\end{proof}

\begin{proposition}[Naturality of the effective divisor map]
  \label{pa-prop-effective-map-natural}
  \dcref{3.7,custom}
  \uses{pa-lem-relative-divisor-base-change,pa-prop-effective-map}
  \cite{kleiman-picard}
  The maps \(C^{(n)}(T)\to\operatorname{Pic}^n_{C/k}(T)\) commute with every
  base change \(T'\to T\).
\end{proposition}

\begin{proof}
  This is the line-bundle identity of
  \ref{pa-lem-relative-divisor-base-change}, together with the definition of
  the relative Picard quotient.
\end{proof}

\begin{definition}[Symmetric powers]
  \label{pa-def-symmetric-power}
  \dcref{III.6,custom}
  \cite{milne-av,kleiman-picard}
  For \(n\ge0\), \(C^{(n)}\) is the quotient of \(C^n\) by the symmetric group.
  Its geometric points are effective divisors of degree \(n\).
\end{definition}

The Hilbert scheme of length-\(n\) subschemes of a smooth curve represents
this functor.  Its local coordinates are the coefficients of a monic
polynomial, so it is smooth of relative dimension \(n\).

\begin{proposition}[Effective divisor map]
  \label{pa-prop-effective-map}
  \dcref{3.7}
  \uses{pa-def-symmetric-power,pa-def-picard}
  \cite[Thm.~3.7]{kleiman-picard}
  The assignment \(D\mapsto\mathcal O_C(D)\) gives a morphism
  \[
    C^{(n)}\longrightarrow\operatorname{Pic}^n_{C/k}
  \]
  whose geometric fibres are complete linear systems.
\end{proposition}

\begin{proof}
  A finite flat effective divisor is Cartier on the smooth relative curve,
  hence determines an invertible sheaf of degree \(n\).  The construction is
  compatible with pullback and factors through the etale Picard sheaf.  Two
  divisors have the same image exactly when their difference is principal,
  which identifies the geometric fibre with the complete linear system.
\end{proof}

\begin{definition}[Abel--Jacobi map]
  \label{pa-def-abel-jacobi}
  \dcref{III.6,custom}
  \uses{pa-def-picard}
  \cite{milne-av,kleiman-picard}
  For \(P_0\in C(k)\), set
  \[
    \iota_{P_0}(Q)=[\mathcal O_C(Q-P_0)].
  \]
  The rigidified diagonal line bundle makes
  \(\iota_{P_0}:C\to\operatorname{Pic}^0_{C/k}\) a morphism with
  \(\iota_{P_0}(P_0)=0\).
\end{definition}

\begin{proof}
  The diagonal divisor on \(C\times C\), corrected by the two fibres through
  \(P_0\), is rigidified along \(P_0\).  Its restriction over \(Q\) is
  \(\mathcal O_C(Q-P_0)\).  The universal property of the relative Picard
  sheaf therefore gives the stated morphism and sends \(P_0\) to the identity.
\end{proof}
