\section{Rings and ring homomorphisms}
\label{am-ch1-rings}
\source{atiyah-macdonald-commutative-algebra:page-0010}\uses{}
\begin{definition}[Ring]
\label{am-def-ring}
\source{atiyah-macdonald-commutative-algebra:page-0010}\uses{}
A ring is a set with an abelian-group addition, an associative multiplication
distributive over addition, commutative multiplication, and an identity $1$.
The zero ring is allowed ($1=0$).  A ring homomorphism preserves addition,
multiplication, and identity; subrings contain the identity.  Homomorphisms
compose, and the identity is the unique element acting as a multiplicative
identity.
\end{definition}

\section{Ideals and quotient rings}
\label{am-ch1-ideals}
\begin{definition}[Ideal and quotient]
\label{am-def-ideal-quotient}
\source{atiyah-macdonald-commutative-algebra:page-0010}\uses{}
An ideal $\mathfrak{a}\subset A$ is an additive subgroup with $A\mathfrak{a}\subseteq\mathfrak{a}$.
The cosets $A/\mathfrak{a}$ have the induced ring structure, and the quotient map
$\varphi:A\to A/\mathfrak{a}$ is surjective.  Write $x\equiv y\pmod{\mathfrak{a}}$ when
$x-y\in\mathfrak{a}$.
\end{definition}
\begin{proposition}[Ideals of a quotient]
\label{am-prop-1-1}
\dcref{1.1}
\source{atiyah-macdonald-commutative-algebra:page-0010}\uses{}
Ideals of $A/\mathfrak{a}$ correspond bijectively and order-preservingly to ideals
$\mathfrak{b}$ of $A$ containing $\mathfrak{a}$, by $\mathfrak{b}\mapsto\varphi^{-1}(\mathfrak{b})$.
\end{proposition}
\begin{proof}
The inverse sends $\mathfrak{b}\subseteq A$ to $\mathfrak{b}/\mathfrak{a}$.  Closure under
addition and multiplication is immediate, and the two composites are the
identity because a coset belongs to $\mathfrak{b}/\mathfrak{a}$ exactly when its
representative belongs to $\mathfrak{b}$.
\end{proof}
\begin{remark}[Kernel and image]
\label{am-rem-kernel-image}
\source{atiyah-macdonald-commutative-algebra:page-0010}\uses{}
For a homomorphism $f:A\to B$, $\Ker f$ is an ideal, $\Im f$ is a subring, and
$f$ induces an isomorphism $A/\Ker f\simeq\Im f$.
\uses{am-def-ring,am-def-ideal-quotient}
\end{remark}

\section{Zero-divisors, nilpotents, and units}
\label{am-ch1-elementary}
\begin{definition}[Domain, nilpotent, unit, principal ideal]
\label{am-def-domain-nil-unit}
\source{atiyah-macdonald-commutative-algebra:page-0011}\uses{}
A zero-divisor is an $x$ for which $xy=0$ for some $y\ne0$; a nonzero ring
without zero-divisors is an integral domain.  An element is nilpotent if
$x^n=0$ for some $n>0$.  A unit has a multiplicative inverse, necessarily
unique, denoted $x^{-1}$; the units form an abelian group.  The principal ideal
generated by $x$ is $(x)=Ax$, and $x$ is a unit iff $(x)=A$.  A field is a
nonzero ring in which every nonzero element is a unit.
\end{definition}
\begin{proposition}[Characterizations of a field]
\label{am-prop-1-2}
\dcref{1.2}
\source{atiyah-macdonald-commutative-algebra:page-0012}\uses{}
For a ring $A\ne0$, the following are equivalent: (i) $A$ is a field; (ii) its
only ideals are $(0)$ and $(1)$; (iii) every homomorphism from $A$ to a nonzero
ring is injective.
\end{proposition}
\begin{proof}
If $A$ is a field and $\mathfrak{a}\ne0$, a nonzero $x\in\mathfrak{a}$ is a unit, so
$1\in\mathfrak{a}$.  If (ii) holds and $f:A\to B\ne0$, then $\Ker f\ne A$ and is
therefore zero.  Conversely, if $x$ is not a unit, $(x)\ne(1)$ and
$A/(x)\ne0$; the quotient map is injective by (iii), forcing $(x)=0$ and
$x=0$.  Thus every nonzero element is a unit.
\end{proof}

\section{Prime and maximal ideals}
\label{am-ch1-prime-max}
\begin{definition}[Prime and maximal ideals]
\label{am-def-prime-max}
\source{atiyah-macdonald-commutative-algebra:page-0012}\uses{}
An ideal $\mathfrak{p}$ is prime if $\mathfrak{p}\ne(1)$ and $xy\in\mathfrak{p}$ implies $x\in\mathfrak{p}$ or
$y\in\mathfrak{p}$.  An ideal $\mathfrak{m}$ is maximal if $\mathfrak{m}\ne(1)$ and no proper ideal lies
strictly between $\mathfrak{m}$ and $(1)$.
\end{definition}
\begin{proposition}[Quotient tests]
\label{am-prop-prime-max-quotient}
\source{atiyah-macdonald-commutative-algebra:page-0012}\uses{}
$\mathfrak{p}$ is prime iff $A/\mathfrak{p}$ is a domain, and $\mathfrak{m}$ is maximal iff $A/\mathfrak{m}$ is a
field.  In particular maximal ideals are prime, and $(0)$ is prime iff $A$ is
an integral domain.  The inverse image of a prime under a homomorphism is
prime, while the inverse image of a maximal ideal need not be maximal.
\uses{am-def-prime-max,am-def-ideal-quotient,am-def-domain-nil-unit}
\end{proposition}
\begin{proof}
The quotient criterion follows by translating products and properness through
the quotient correspondence (Proposition~\ref{am-prop-1-1}) and the field
criterion (Proposition~\ref{am-prop-1-2}).  If $f:A\to B$ and $\mathfrak{q}$ is prime,
then $A/f^{-1}(\mathfrak{q})$ embeds in the domain $B/\mathfrak{q}$, hence is a domain.
\end{proof}
\begin{theorem}[Existence of maximal ideals]
\label{am-thm-1-3}
\dcref{1.3}
\source{atiyah-macdonald-commutative-algebra:page-0013}\uses{}
Every nonzero ring has a maximal ideal.
\end{theorem}
\begin{proof}
Order the proper ideals by inclusion.  The union of a chain is an ideal and
cannot contain $1$, hence is an upper bound in this poset.  Zorn's lemma gives
a maximal member.
\end{proof}
\begin{corollary}[Maximal ideal over a proper ideal]
\label{am-cor-1-4}
\dcref{1.4}
\source{atiyah-macdonald-commutative-algebra:page-0014}\uses{}
Every proper ideal is contained in a maximal ideal.
\uses{am-prop-1-1,am-thm-1-3}
\end{corollary}
\begin{proof}
Apply Theorem~\ref{am-thm-1-3} to $A/\mathfrak{a}$ and pull the maximal ideal back
using Proposition~\ref{am-prop-1-1}.
\end{proof}
\begin{corollary}[Nonunits lie in maximal ideals]
\label{am-cor-1-5}
\dcref{1.5}
\source{atiyah-macdonald-commutative-algebra:page-0014}\uses{}
Every nonunit belongs to a maximal ideal.
\uses{am-cor-1-4,am-def-domain-nil-unit}
\end{corollary}
\begin{proof}
The principal ideal generated by a nonunit is proper; apply Corollary~\ref{am-cor-1-4}.
\end{proof}
\begin{definition}[Local and semilocal rings]
\label{am-def-local-ring}
\source{atiyah-macdonald-commutative-algebra:page-0014}\uses{}
A ring with one maximal ideal $\mathfrak{m}$ is local and $A/\mathfrak{m}$ is its residue field.  A
ring with finitely many maximal ideals is semilocal.
\end{definition}
\begin{proposition}[Tests for locality]
\label{am-prop-1-6}
\dcref{1.6}
\source{atiyah-macdonald-commutative-algebra:page-0014}\uses{}
If $\mathfrak{m}\ne(1)$ and every element of $A\setminus\mathfrak{m}$ is a unit, then $A$ is local
with maximal ideal $\mathfrak{m}$.  If $\mathfrak{m}$ is maximal and every element of $1+\mathfrak{m}$ is a
unit, then $A$ is local.
\end{proposition}
\begin{proof}
In the first case every proper ideal consists of nonunits and is contained in
$\mathfrak{m}$.  In the second, for $x\notin\mathfrak{m}$, maximality gives
$xy+t=1$ with $t\in\mathfrak{m}$, so $xy=1-t$ is a unit; hence $x$ is a unit
and the first case applies.
\end{proof}

\section{Nilradical and Jacobson radical}
\label{am-ch1-radicals}
\begin{proposition}[The nilradical]
\label{am-prop-1-7}
\dcref{1.7}
\source{atiyah-macdonald-commutative-algebra:page-0015}\uses{}
The set $\Nil(A)$ of nilpotent elements is an ideal, and $A/\Nil(A)$ has no
nonzero nilpotent element.
\end{proposition}
\begin{proof}
If $x^m=y^n=0$, the binomial expansion of $(x+y)^{m+n-1}$ has every monomial
zero, since each term has exponent at least $m$ in $x$ or at least $n$ in $y$.
Scalar multiples are nilpotent.  If $\bar x^r=0$ in the quotient, then
$x^r$ is nilpotent, hence $x$ is nilpotent and $\bar x=0$.
\end{proof}
\begin{proposition}[Nilradical as an intersection]
\label{am-prop-1-8}
\dcref{1.8}
\source{atiyah-macdonald-commutative-algebra:page-0015}\uses{}
The nilradical of $A$ is the intersection of all prime ideals of $A$.
\end{proposition}
\begin{proof}
Every nilpotent lies in every prime ideal.  If $f$ is not nilpotent, consider
the ideals $\mathfrak{a}$ such that $f^n\notin\mathfrak{a}$ for all $n>0$.  A maximal such
ideal $\mathfrak{p}$ exists by the same chain-union argument as in Theorem~\ref{am-thm-1-3}.
If $x,y\notin\mathfrak{p}$, maximality gives $f^m\in\mathfrak{p}+(x)$ and
$f^n\in\mathfrak{p}+(y)$; multiplying and collecting terms gives
$f^{m+n}\in\mathfrak{p}+(xy)$.  Thus $xy\notin\mathfrak{p}$, so $\mathfrak{p}$ is prime and does not contain
$f$.  Therefore every nonnilpotent is omitted by some prime.
\end{proof}
\begin{definition}[Jacobson radical]
\label{am-def-jacobson}
\source{atiyah-macdonald-commutative-algebra:page-0015}\uses{}
The Jacobson radical $J(A)$ is the intersection of all maximal ideals of $A$.
\end{definition}
\begin{proposition}[Unit characterization of the Jacobson radical]
\label{am-prop-1-9}
\dcref{1.9}
\source{atiyah-macdonald-commutative-algebra:page-0016}\uses{}
$x\in J(A)$ iff $1-xy$ is a unit for every $y\in A$.
\end{proposition}
\begin{proof}
If $1-xy$ were a nonunit, Corollary~\ref{am-cor-1-5} would put it in a maximal
ideal containing $x$, giving $1\in\mathfrak{m}$.  Conversely, if $x\notin\mathfrak{m}$ for a
maximal $\mathfrak{m}$, write $u+xy=1$ with $u\in\mathfrak{m}$; then $1-xy=u$ is not a unit.
\uses{am-def-jacobson,am-cor-1-5}
\end{proof}

\section{Operations on ideals}
\label{am-ch1-operations}
\begin{definition}[Ideal operations]
\label{am-def-ideal-operations}
\source{atiyah-macdonald-commutative-algebra:page-0016}\uses{}
For ideals $\mathfrak{a},\mathfrak{b}$, their sum is the ideal of finite sums, their
intersection is set-theoretic intersection, and their product $\mathfrak{a}\mathfrak{b}$
is generated by finite sums of products $xy$ with $x\in\mathfrak{a},y\in\mathfrak{b}$.
Powers use $\mathfrak{a}^0=(1)$.  The ideal quotient is
$(\mathfrak{a}:\mathfrak{b})=\{x:x\mathfrak{b}\subseteq\mathfrak{a}\}$; $(0:\mathfrak{b})$ is
$\Ann(\mathfrak{b})$.  The radical is
$\rad(\mathfrak{a})=\{x:x^n\in\mathfrak{a}\text{ for some }n>0\}$.
\end{definition}
\begin{remark}[Lattice identities]
\label{am-rem-ideal-identities}
\source{atiyah-macdonald-commutative-algebra:page-0016}\uses{}
Ideal sum, intersection, and product are associative and commutative, product
distributes over sum, and the modular law
$\mathfrak{a}\cap(\mathfrak{b}+\mathfrak{c})=\mathfrak{b}+(\mathfrak{a}\cap\mathfrak{c})$
holds when $\mathfrak{a}\supseteq\mathfrak{b}$ (with the symmetric form when
$\mathfrak{a}\supseteq\mathfrak{c}$).  If
$\mathfrak{a}+\mathfrak{b}=(1)$, then $\mathfrak{a}\cap\mathfrak{b}=\mathfrak{a}\mathfrak{b}$.
\end{remark}
\begin{proposition}[Chinese remainder theorem]
\label{am-prop-1-10}
\dcref{1.10}
\source{atiyah-macdonald-commutative-algebra:page-0017}\uses{}
For ideals $\mathfrak{a}_1,\ldots,\mathfrak{a}_n$, define
$\varphi:A\to\prod_i A/\mathfrak{a}_i$.  If the ideals are pairwise coprime, then
$\bigcap_i\mathfrak{a}_i=\prod_i\mathfrak{a}_i$ and $\varphi$ is surjective.  In general
$\varphi$ is injective iff $\bigcap_i\mathfrak{a}_i=(0)$, and it is surjective iff
the ideals are pairwise coprime.
\end{proposition}
\begin{proof}
For two coprime ideals, write $1=x+y$ with $x\in\mathfrak{a},y\in\mathfrak{b}$ to
obtain $\mathfrak{a}\cap\mathfrak{b}=\mathfrak{a}\mathfrak{b}$ and solve the two congruences.
Inductively, the product of the first $n-1$ ideals is coprime to the last,
and the same construction gives a preimage of each coordinate tuple.  The
kernel of $\varphi$ is the intersection, proving injectivity.  Surjectivity
implies pairwise coprimality by applying it to the tuple with coordinates
$1,0,\ldots,0$; the converse is the explicit product construction.
\uses{am-def-ideal-quotient,am-def-ideal-operations}
\end{proof}
\begin{proposition}[Prime avoidance]
\label{am-prop-1-11}
\dcref{1.11}
\source{atiyah-macdonald-commutative-algebra:page-0018}\uses{}
If an ideal is contained in a finite union of prime ideals, it is contained in
one of them.  If a prime ideal contains a finite intersection of ideals, it
contains one of those ideals; if it equals the intersection, it equals one of
them.
\end{proposition}
\begin{proof}
Induct on the number of primes.  If no element of the ideal separates one
prime from all the others, choose $x_i$ in the ideal lying in every prime but
$\mathfrak{p}_i$; the product of the $x_i$ lies in the ideal but in no $\mathfrak{p}_i$, a
contradiction.  For the second assertion, if no $\mathfrak{a}_i$ is contained in
the prime, choose $x_i\in\mathfrak{a}_i\setminus\mathfrak{p}$; their product lies in the
intersection but not in $\mathfrak{p}$.
\end{proof}
\begin{proposition}[Radical as an intersection of primes]
\label{am-prop-1-14}
\dcref{1.14}
\source{atiyah-macdonald-commutative-algebra:page-0018}\uses{}
$\rad(\mathfrak{a})$ is the intersection of the prime ideals containing $\mathfrak{a}$.
\end{proposition}
\begin{proof}
Apply Proposition~\ref{am-prop-1-8} to $A/\mathfrak{a}$ for the first assertion.
\uses{am-prop-1-8,am-def-ideal-operations}
\end{proof}
\begin{definition}[Annihilator of an element]
\label{am-def-ann-element}
\source{atiyah-macdonald-commutative-algebra:page-0018}\uses{}
For $x\in A$, its annihilator is the ideal
$\Ann(x)=(0:(x))=\{a\in A:ax=0\}$.
\end{definition}
\begin{proposition}[Zero-divisors from annihilators]
\label{am-prop-1-15}
\dcref{1.15}
\source{atiyah-macdonald-commutative-algebra:page-0018}\uses{}
If $D$ is the set of zero-divisors, then
$D=\bigcup_{x\ne0}\rad(\Ann(x))$.
\uses{am-prop-1-14,am-def-ann-element}
\end{proposition}
\begin{proof}
$D=\bigcup_{x\ne0}\Ann(x)$ by definition, and radicals commute with unions
of subsets in the sense $\rad(\bigcup E_i)=\bigcup\rad(E_i)$.
\end{proof}
\begin{proposition}[Coprime radicals]
\label{am-prop-1-16}
\dcref{1.16}
\source{atiyah-macdonald-commutative-algebra:page-0018}\uses{}
If $\rad(\mathfrak{a})$ and $\rad(\mathfrak{b})$ are coprime, then
$\mathfrak{a}+\mathfrak{b}=(1)$.
\uses{am-def-ideal-operations,am-prop-1-14}
\end{proposition}
\begin{proof}
$\rad(\mathfrak{a}+\mathfrak{b})=\rad(\rad(\mathfrak{a})+\rad(\mathfrak{b}))=(1)$, so
$\mathfrak{a}+\mathfrak{b}=(1)$.
\end{proof}

\section{Extension and contraction}
\label{am-ch1-extension-contraction}

\begin{definition}[Extension and contraction]
\label{am-def-extension-contraction}
\source{atiyah-macdonald-commutative-algebra:page-0018}\uses{}
For a homomorphism $f:A\to B$ and ideals $\mathfrak{a}\subseteq A$ and
$\mathfrak{b}\subseteq B$, write
$\mathfrak{a}^e=Bf(\mathfrak{a})$ for the ideal generated by $f(\mathfrak{a})$ and
$\mathfrak{b}^c=f^{-1}(\mathfrak{b})$ for its contraction.  These operations are
order-preserving; extension is used for ideals of $A$ and contraction for ideals
of $B$.  The factorization $A\twoheadrightarrow f(A)\hookrightarrow B$ reduces
the surjective case to the quotient correspondence.
\end{definition}

\begin{proposition}[Extension and contraction]
\label{am-prop-1-17}
\dcref{1.17}
\source{atiyah-macdonald-commutative-algebra:page-0019}\uses{}
For ideals $\mathfrak{a}\subseteq A$ and $\mathfrak{b}\subseteq B$,
$\mathfrak{a}\subseteq\mathfrak{a}^{ec}$ and $\mathfrak{b}\supseteq\mathfrak{b}^{ce}$.
Moreover $\mathfrak{b}^c=\mathfrak{b}^{cec}$ and $\mathfrak{a}^e=\mathfrak{a}^{ece}$.
Contraction and extension give inverse bijections between contracted ideals and
extended ideals.
\uses{am-def-extension-contraction,am-prop-1-1}
\end{proposition}
\begin{proof}
The first inclusions follow from $f(\mathfrak{a})\subseteq\mathfrak{b}$ whenever
$\mathfrak{a}\subseteq\mathfrak{b}^c$, and from taking preimages of generated ideals.
Applying these inclusions twice gives the equalities for already contracted or
extended ideals.  Thus the fixed points of the two closure operations are in
bijective correspondence, with the stated maps inverse.
\end{proof}
