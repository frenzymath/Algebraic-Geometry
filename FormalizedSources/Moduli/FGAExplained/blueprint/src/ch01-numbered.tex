\begin{remark}[local affine covers do not detect quasi-compactness]\label{fga-rem-1-5}\dcref{1.5}\source{fga-explained:page-0018}\uses{fga-def-1-3,fga-prop-1-4}
It is not sufficient to cover $Y$ by open quasi-compact subschemes $V_i$ for
which every $f^{-1}(V_i)$ is quasi-compact.  For example, let
$X=\operatorname{Spec}A$ contain an open subset $U$ which is not quasi-compact,
and form $Y$ by gluing two copies of $X$ along $U$.  The inclusion of either
copy $X\to Y$ has quasi-compact source and target, but the inverse image of the
other affine copy is $U$; hence the inclusion is not quasi-compact.
\end{remark}

\begin{definition}[faithfully flat morphism and algebra]\label{fga-def-1-10}\dcref{1.10}\source{fga-explained:page-0019}\uses{fga-def-1-7}
A morphism of schemes $f:X\to Y$ is faithfully flat if it is flat and
surjective.  An $A$-algebra $B$ is faithfully flat if the associated morphism
$\operatorname{Spec}B\to\operatorname{Spec}A$ is faithfully flat.
\end{definition}

\begin{remark}[quasi-compactness in descent of openness]\label{fga-rem-1-14}\dcref{1.14}\source{fga-explained:page-0019}\uses{fga-prop-1-13}
The quasi-compactness hypothesis in descent of openness cannot be replaced by
faithful flatness alone.  If $Y$ is a smooth integral curve over an
algebraically closed field and
\[
X=\coprod_{y\in Y_{\mathrm{cl}}}\operatorname{Spec}\mathcal O_{Y,y},
\]
the natural morphism $X\to Y$ is faithfully flat.  For a subset $A\subset Y$
containing the generic point, its inverse image is open in $X$, whereas $A$ is
open in $Y$ only when its complement is finite.
\end{remark}
