\chapter{Rigidity and the Albanese property}
\label{pa-ch-albanese}

\begin{theorem}[Rigidity]
  \label{pa-thm-rigidity}
  \dcref{I.2}
  \cite{milne-av,mumford-av}
  If \(X\) is proper and geometrically connected, \(Y\) is reduced, \(A\) is
  an abelian variety, and \(f:X\times_kY\to A\) is constant on
  \(\{x_0\}\times Y\), then \(f\) factors uniquely through \(X\).  In
  particular a pointed morphism from an abelian variety to \(A\) is a
  homomorphism.
\end{theorem}

\begin{proof}
  Extend scalars to an algebraic closure and choose an affine neighbourhood of
  the common value on \(\{x_0\}\times Y\).  Properness makes the locus of
  \(x\) for which the \(Y\)-fibre leaves this
  neighbourhood closed.  On the complement, regular functions pulled back
  from the affine target are constant in the \(Y\)-direction; the value at
  \(x_0\) fixes the constants.
  The equalizer is closed because \(A\) is separated and contains a dense open
  of the reduced product, hence it is the whole product.  Apply this to
  \(f(x+y)-f(x)-f(y)\) for the homomorphism assertion.
\end{proof}

\begin{lemma}[Rational graph closure]
  \label{pa-lem-rational-graph}
  \dcref{3.1}
  \cite{milne-av}
  If \(A\) is proper, the graph closure of a rational map \(X\dashrightarrow A\)
  is proper and birational over \(X\).
\end{lemma}

\begin{proof}
  It is closed in \(X\times A\), so its projection is proper; over the dense
  domain of definition it is the original graph.
\end{proof}

\begin{lemma}[Codimension-one points give DVRs]
  \label{pa-lem-codim-one-dvr}
  \dcref{3.2}
  \cite{milne-av,hartshorne-ag}
  A codimension-one point of a nonsingular variety has a DVR as local ring.
\end{lemma}

\begin{proof}
  Its local ring is a one-dimensional regular local domain, hence a DVR.
\end{proof}

\begin{lemma}[Valuative extension at codimension one]
  \label{pa-lem-valuative-extension}
  \dcref{3.2}
  \uses{pa-lem-rational-graph,pa-lem-codim-one-dvr}
  \cite{milne-av}
  A rational map from a nonsingular variety to a proper variety is defined at
  every codimension-one point.
\end{lemma}

\begin{proof}
  Apply the valuative criterion for properness to the DVR in
  \ref{pa-lem-codim-one-dvr}; uniqueness gives a single point of the graph.
\end{proof}

\begin{definition}[Rational map and domain]
  \label{pa-def-rational-map}
  \dcref{3.1}
  \cite{milne-av}
  A rational map \(X\dashrightarrow Y\) is an equivalence class of morphisms
  \(U\to Y\) with \(U\subset X\) dense open; its maximal such \(U\) is the
  domain of definition.
\end{definition}

\begin{lemma}[Rational maps are determined generically]
  \label{pa-lem-rational-generic}
  \dcref{3.1}
  \uses{pa-def-rational-map}
  \cite{milne-av}
  If \(X\) is integral and \(Y\) separated, two rational maps \(X\dashrightarrow
  Y\) agreeing at the generic point agree on the intersection of their domains.
\end{lemma}

\begin{proof}
  Their equalizer is closed in the intersection because \(Y\) is separated and
  contains the generic point.  The intersection is integral, so the equalizer
  contains a dense open and the maps define the same rational class.
\end{proof}

\begin{lemma}[Empty indeterminacy gives a morphism]
  \label{pa-lem-empty-indeterminacy}
  \dcref{3.2}
  \uses{pa-lem-rational-graph,pa-lem-valuative-extension}
  \cite{milne-av}
  If the graph closure of \(f:X\dashrightarrow Y\) has singleton fibres over
  all points of \(X\), then \(f\) extends uniquely to a morphism.
\end{lemma}

\begin{proof}
  The graph projection is proper and quasi-finite, hence finite.  A finite
  birational morphism with singleton fibres to a normal scheme is an
  isomorphism; its inverse is the required extension.
\end{proof}

\begin{lemma}[The Abel factor is a homomorphism]
  \label{pa-lem-abel-factor-homomorphism}
  \dcref{III.6}
  \uses{pa-thm-albanese,pa-thm-rigidity}
  \cite{milne-av}
  The morphism \(u:\operatorname{Pic}^0_{C/k}\to A\) obtained from the
  symmetric-power construction respects addition.
\end{lemma}

\begin{proof}
  Compare the constructions for degrees \(n\) and \(n+1\), using the divisor
  \(P_0\) to embed \(C^{(n)}\) into \(C^{(n+1)}\).  The resulting equality on
  the surjective Abel maps gives \(u(L\otimes M)=u(L)+u(M)\).
\end{proof}

\begin{lemma}[Symmetric sums]
  \label{pa-lem-symmetric-sum}
  \dcref{III.6}
  \uses{pa-def-abel-jacobi}
  \cite{milne-av}
  For \(f:C\to A\), the sum \(f(Q_1)+\cdots+f(Q_n)\) descends to a morphism
  \(C^{(n)}\to A\).
\end{lemma}

\begin{proof}
  The sum is invariant under \(S_n\) because \(A\) is commutative.  The
  quotient property of \(C^{(n)}\) gives the descended morphism.
\end{proof}

\begin{lemma}[Constancy on complete linear systems]
  \label{pa-lem-linear-system-constant}
  \dcref{I.2}
  \uses{pa-lem-effective-map-fibres,pa-thm-rigidity}
  \cite{milne-av}
  A morphism from a complete linear system to an abelian variety is constant.
\end{lemma}

\begin{proof}
  A complete linear system is a projective space.  Its image is complete and
  connected, and the rigidity lemma forces a map from this projective space to
  an abelian variety to be constant.
\end{proof}

\begin{theorem}[Extension of rational maps]
  \label{pa-thm-rational-extension}
  \dcref{3.1,3.2}
  \uses{pa-thm-rigidity}
  \cite[Thms.~3.1--3.2]{milne-av}
  A rational map from a nonsingular variety to an abelian variety extends
  uniquely to a morphism.
\end{theorem}

\begin{proof}
  Take the closure of the graph in \(X\times A\).  Since \(A\) is proper, the
  projection to \(X\) is proper and birational.  At every codimension-one
  point of the nonsingular \(X\), the local ring is a discrete valuation ring,
  and the valuative criterion gives a unique extension there.  The remaining
  indeterminacy has codimension at least two; the rigidity lemma forces every
  fibre of the graph projection to be a singleton.  The graph projection is
  therefore an isomorphism and supplies the unique morphism extending the
  rational map.
\end{proof}

\begin{theorem}[Albanese universal property]
  \label{pa-thm-albanese}
  \dcref{III.6}
  \uses{pa-def-abel-jacobi,pa-thm-pic0-abelian,pa-thm-rigidity}
  \cite{milne-av}
  For every abelian variety \(A/k\) and every morphism
  \(f:C\to A\) with \(f(P_0)=0\), there is a unique homomorphism
  \(u:\operatorname{Pic}^0_{C/k}\to A\) satisfying
  \[
    f=u\circ\iota_{P_0}.
  \]
\end{theorem}

\begin{proof}
  Choose \(n\ge 2g(C)-1\).  The sum of the values of \(f\) on an ordered
  \(n\)-tuple is symmetric, hence descends to \(f_n:C^{(n)}\to A\).  The
  fibres of \(C^{(n)}\to\operatorname{Pic}^n_{C/k}\) are complete linear
  systems, and the rigidity theorem makes \(f_n\) constant on each fibre.
  It therefore descends to \(\operatorname{Pic}^n_{C/k}\).  Translation by a
  degree-\(n\) line bundle gives a morphism
  \(u:\operatorname{Pic}^0_{C/k}\to A\).  Comparing \(n\) and \(n+1\) shows
  \(u\) respects addition and that \(f=u\circ\iota_{P_0}\).  Surjectivity of
  the symmetric-power map gives uniqueness.
\end{proof}
