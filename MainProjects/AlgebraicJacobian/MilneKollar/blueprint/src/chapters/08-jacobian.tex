\chapter{The Jacobian}
\label{mk-ch-jacobian}

\begin{definition}[Degree-zero Picard functor]
  \label{mk-def-pic0-functor}
  \dcref{5.1,custom}
  \uses{mk-def-picard-functor,mk-lem-degree-open-closed}
  \cite{kleiman-picard}
  For a \(k\)-scheme \(T\), \(\operatorname{Pic}^0_{C/k}(T)\) is the
  identity component of the represented relative Picard group after base
  change to \(T\).
\end{definition}

\begin{lemma}[Degree-zero classes are stable under base change]
  \label{mk-lem-pic0-base-change}
  \dcref{5.1,custom}
  \uses{mk-def-pic0-functor,mk-lem-cech-flat-cohomology}
  \cite{kleiman-picard}
  For every field extension \(k\subset k'\),
  \[
    (\operatorname{Pic}^0_{C/k})_{k'}\cong
    \operatorname{Pic}^0_{C_{k'}/k'}.
  \]
\end{lemma}

\begin{proof}
  Base change preserves the tensor law and the identity section.  By
  \ref{mk-lem-degree-open-closed}, the degree-zero locus is open and closed;
  the connected component containing the identity is therefore carried to
  the corresponding component after base change.
\end{proof}

\begin{proposition}[Functoriality for morphisms of pointed curves]
  \label{mk-prop-jacobian-functorial}
  \dcref{III.6,custom}
  \uses{mk-def-jacobian,mk-def-abel-jacobi}
  \cite{milne-av}
  A morphism \(u:C\to C'\) with \(u(P_0)=P'_0\) induces
  \(u^*:\operatorname{Jac}(C')\to\operatorname{Jac}(C)\), and
  \(u^*\circ\iota_{P'_0}=\iota_{P_0}\circ u\).
\end{proposition}

\begin{proof}
  Pullback preserves tensor products, inverses, and degree, so it restricts to
  the identity components.  Pulling back
  \(\mathcal O_{C'}(Q'-P'_0)\) gives
  \(\mathcal O_C(u^{-1}Q'-P_0)\); the diagonal description gives the stated
  compatibility.
\end{proof}

\begin{lemma}[The Jacobian has a canonical zero]
  \label{mk-lem-jacobian-zero}
  \dcref{5.1,custom}
  \uses{mk-def-jacobian}
  \cite{kleiman-picard}
  The trivial line bundle defines a \(k\)-rational identity point of
  \(\operatorname{Jac}(C)\), independent of all choices.
\end{lemma}

\begin{proof}
  The identity section of the Picard group is represented by
  \(\mathcal O_C\); its degree is zero and the tensor law makes it the neutral
  element of the identity component.
\end{proof}

\begin{definition}[Jacobian]
  \label{mk-def-jacobian}
  \dcref{III.6,custom}
  \uses{mk-thm-pic0-abelian}
  \cite{milne-av,kleiman-picard}
  The Jacobian of \(C/k\) is
  \[
    \operatorname{Jac}(C):=\operatorname{Pic}^0_{C/k}.
  \]
\end{definition}

\begin{remark}
  The definition is independent of a chosen rational point.  A point is used
  only to write the Abel--Jacobi map explicitly; changing it translates that
  map.
\end{remark}

\begin{theorem}[Jacobian package]
  \label{mk-thm-jacobian-package}
  \dcref{III.6,5.13,5.14,custom}
  \uses{mk-def-jacobian,mk-thm-abel-jacobi-universal,mk-thm-pic0-abelian}
  \cite{milne-av,kleiman-picard}
  By \ref{mk-thm-pic0-abelian}, the scheme \(\operatorname{Jac}(C)\) is an
  abelian variety of dimension \(g(C)\).  The map \(\iota_{P_0}\) is independent
  of the chosen base point up to translation, and it satisfies the universal
  property in \ref{mk-thm-abel-jacobi-universal}.
\end{theorem}

\begin{proof}
  The Picard-variety theorem identifies \(\operatorname{Pic}^0_{C/k}\) with
  an abelian variety and computes its dimension as \(g(C)\).  The
  Abel--Jacobi universal property then gives the stated factorisation for
  every pointed map from \(C\) to an abelian variety.  If \(P_1\) replaces
  \(P_0\), the two maps differ by the translation
  \(Q\mapsto Q-[\mathcal O_C(P_1-P_0)]\), by the tensor law.
\end{proof}

\begin{theorem}[Base change]
  \label{mk-thm-jacobian-base-change}
  \dcref{III.6,custom}
  \uses{mk-def-jacobian,mk-thm-picard-representability}
  For every field extension \(k\subset k'\),
  \[
    \operatorname{Jac}(C)\times_k k'\cong
    \operatorname{Jac}(C\times_k k').
  \]
  The isomorphism is canonical and compatible with towers of extensions and
  with Abel--Jacobi maps.
\end{theorem}

\begin{proof}
  Picard representability and the cohomology-and-base-change theorem identify
  the base change of each degree component with the corresponding component
  for \(C_{k'}\).  The identity section and tensor law are preserved by this
  identification, so the identity components agree.  Restricting to degree
  zero gives the canonical isomorphism of Jacobians, and the construction of
  the diagonal line bundle shows compatibility with Abel--Jacobi maps.
\end{proof}
