\chapter{Line-bundle pullback on a relative curve}
\label{chap:Picard_LineBundlePullback}
% archon:covers AlgebraicJacobian/Picard/LineBundlePullback.lean

Let $S$ be a scheme, let $C \to S$ be a morphism, and let $T \to S$ be a
test scheme.  Write $P=C\times_S T$ for their fibre product.  When $S=\Spec k$
and $C$ is a smooth proper geometrically integral curve, this is the relative
curve used in the Picard construction.  We first record the local description
of line bundles needed for pullback.

\section{Line bundles and local triviality}
\label{sec:lbp_line_bundle}

\begin{definition}[Local triviality of a module]\leanok
  \label{def:IsLocallyTrivial}
  \lean{AlgebraicGeometry.Scheme.LineBundle.IsLocallyTrivial}
  \dcref{custom}
  An $\mathcal O_X$-module $\mathcal M$ on a scheme $X$ is \emph{locally
  trivial of rank one} if every point $x\in X$ has an affine open
  neighbourhood $U$ and an isomorphism
  \[
    \mathcal M|_U \;\cong\; \mathcal O_U.
  \]
  Equivalently, in the category of sheaves of $\mathcal O_X$-modules,
  \[
    \forall x:X,\;\exists U:X.\mathrm{Opens},\quad
    x\in U\;\wedge\;U\text{ affine}\;\wedge\;
    \mathcal M|_U\cong\mathcal O_U.
  \]
\end{definition}

\begin{definition}[Line bundle on a product]\leanok
  \label{def:line_bundle_on_product}
  \lean{AlgebraicGeometry.Scheme.LineBundle.OnProduct}
  \uses{def:IsLocallyTrivial}
  \dcref{custom}
  A \emph{line bundle on $P=C\times_S T$} is a locally trivial module of rank
  one on $P$.  Thus the type of line bundles on the product is
  \[
    \mathrm{OnProduct}(C\to S,T\to S)
    :=\{\mathcal M\in P.\mathrm{Modules}\mid
          \mathcal M\text{ is locally trivial of rank one}\}.
  \]
  Isomorphism classes of these objects are the usual elements of
  $\Pic(P)$; the tensor-product group structure is not needed for the
  constructions below.
\end{definition}

\begin{lemma}[Pullback preserves local triviality]\leanok
  \label{lem:IsLocallyTrivial_pullback}
  \lean{AlgebraicGeometry.Scheme.LineBundle.IsLocallyTrivial.pullback}
  \uses{def:IsLocallyTrivial}
  \dcref{custom}
  Let $f:Y\to X$ be a morphism of schemes and let $\mathcal M$ be locally
  trivial of rank one on $X$.  Then $f^*\mathcal M$ is locally trivial of rank
  one on $Y$.
\end{lemma}
\begin{proof}
  \uses{def:IsLocallyTrivial}
  \leanok
  Fix $y\in Y$ and put $x=f(y)$.  Choose an affine open $U$ containing $x$
  and an isomorphism $\mathcal M|_U\cong\mathcal O_U$.  The inverse image
  $f^{-1}(U)$ is an open neighbourhood of $y$; choose an affine open
  $V\subseteq f^{-1}(U)$ containing $y$.  The map $V\to X$ factors through
  $U$, and restriction followed by pullback gives
  \[
    (f^*\mathcal M)|_V
      \cong (V\to U)^*(\mathcal M|_U)
      \cong (V\to U)^*\mathcal O_U
      \cong \mathcal O_V.
  \]
  Hence every point of $Y$ has an affine trivialising neighbourhood.
\end{proof}
\begin{definition}[Underlying module]\leanok
  \label{def:OnProduct_carrier}
  \dcref{custom}
  \lean{AlgebraicGeometry.Scheme.LineBundle.OnProduct.carrier}
  \uses{def:line_bundle_on_product, def:IsLocallyTrivial}
  For a line bundle $\mathcal L\in\mathrm{OnProduct}(C\to S,T\to S)$, its
  \emph{carrier} $\mathcal L.\mathrm{carrier}$ is the underlying
  $\mathcal O_P$-module obtained by forgetting the local-triviality witness.
\end{definition}

\begin{lemma}[A line bundle is locally trivial]\leanok
  \label{lem:OnProduct_isLocallyTrivial}
  \dcref{custom}
  \lean{AlgebraicGeometry.Scheme.LineBundle.OnProduct.isLocallyTrivial}
  \uses{def:line_bundle_on_product, def:OnProduct_carrier,
    def:IsLocallyTrivial}
  If $\mathcal L\in\mathrm{OnProduct}(C\to S,T\to S)$, then
  $\mathcal L.\mathrm{carrier}$ is locally trivial of rank one.
\end{lemma}
\begin{proof}
  \uses{def:IsLocallyTrivial}
  \leanok
  An element of $\mathrm{OnProduct}$ is a pair consisting of a module and a
  local-triviality witness.  Projecting to the first component leaves exactly
  that witness, so the carrier has the asserted property.
\end{proof}

\section{Pullback along the projection}
\label{sec:lbp_pullback_projection}

Let $\pi_T:P\to T$ be the second projection.  Pullback along $\pi_T$ sends a
line bundle on $T$ to a line bundle on $P$.

\begin{definition}[Pullback along the projection]\leanok
  \label{def:pullback_along_projection}
  \lean{AlgebraicGeometry.Scheme.LineBundle.pullbackAlongProjection}
  \uses{def:line_bundle_on_product, lem:IsLocallyTrivial_pullback}
  \dcref{custom}
  If $\mathcal N$ is a locally trivial $\mathcal O_T$-module of rank one, define
  \[
    \pi_T^*\mathcal N
      :=\bigl((\pi_T)^*\mathcal N,\;\text{the local-triviality witness
         supplied by pullback preservation}\bigr)
      \in \mathrm{OnProduct}(C\to S,T\to S).
  \]
  On isomorphism classes this is the usual pullback operation on line bundles.
\end{definition}

\section{Base change and composition}
\label{sec:lbp_compose}

\begin{lemma}[Composition of line-bundle pullbacks]\leanok
  \label{lem:pullback_compose}
  \lean{AlgebraicGeometry.Scheme.LineBundle.pullback_pullback_eq}
  \uses{def:line_bundle_on_product, def:pullback_along_projection}
  \dcref{custom}
  Let $g:T'\to T$ be a morphism over $S$, so that
  $\pi_{T'}=\pi_T\circ g$.  Let
  \[
    g_C:C\times_S T'\longrightarrow C\times_S T
  \]
  be the induced base-change morphism.  For every $\mathcal O_T$-module
  $\mathcal N$ there exists an isomorphism
  \[
    g_C^*\pi_T^*\mathcal N
      \;\cong\;\pi_{T'}^*g^*\mathcal N.
  \]
\end{lemma}
\begin{proof}
  \leanok
  The fibre-product projections satisfy
  $\pi_T\circ g_C=g\circ\pi_{T'}$.  Pullback of modules is compatible with
  composition, so both sides are isomorphic to the pullback of $\mathcal N$
  along this common composite.  Composing these isomorphisms gives the result.
\end{proof}

\section{Isomorphism classes and their naturality}
\label{sec:lbp_relpic_quotient}

The current line-bundle carrier records modules together with a local
trivialisation.  Before adding tensor products and quotienting by bundles from
the base, we can already form its set of isomorphism classes.

\begin{theorem}[Isomorphism classes of line bundles]\leanok
  \label{thm:relative_pic_quotient_well_defined}
  \dcref{custom}
  \lean{AlgebraicGeometry.Scheme.RelPicPresheaf.preimage_subgroup}
  \uses{def:line_bundle_on_product, def:OnProduct_carrier,
    lem:OnProduct_isLocallyTrivial}
  For fixed $C\to S$ and $T\to S$, define
  \[
    \mathcal L\sim\mathcal L'
      \quad\Longleftrightarrow\quad
      \mathcal L.\mathrm{carrier}\cong\mathcal L'.\mathrm{carrier}.
  \]
  This is an equivalence relation on
  $\mathrm{OnProduct}(C\to S,T\to S)$, and the quotient set
  \[
    \mathrm{LB}(C\times_S T):=
    \mathrm{OnProduct}(C\to S,T\to S)/\!\sim
  \]
  is the set of isomorphism classes of line bundles on $C\times_S T$.  The
  canonical projection onto this quotient is surjective.
\end{theorem}
\begin{proof}
  \leanok
  Reflexivity follows from the identity isomorphism, symmetry from taking the
  inverse of an isomorphism, and transitivity from composition.  Thus $\sim$ is
  an equivalence relation, so its quotient exists; the quotient projection is
  surjective by definition.
\end{proof}

\section{Naturality in the test scheme}
\label{sec:lbp_pullback_natural}

\begin{theorem}[Pullback descends to isomorphism classes]\leanok
  \label{thm:pullback_natural}
  \dcref{custom}
  \lean{AlgebraicGeometry.Scheme.RelPicPresheaf.functorial}
  \uses{def:line_bundle_on_product, def:pullback_along_projection,
    lem:IsLocallyTrivial_pullback, lem:pullback_compose,
    thm:relative_pic_quotient_well_defined}
  Let $g:T'\to T$ be a morphism over $S$ and let $g_C$ be its base change along
  $C\to S$.  Pullback along $g_C$ sends line bundles on $C\times_S T$ to line
  bundles on $C\times_S T'$ and sends isomorphic carriers to isomorphic
  carriers.  Consequently it induces a unique map of quotient sets
  \[
    g^\sharp:\mathrm{LB}(C\times_S T)\longrightarrow
                    \mathrm{LB}(C\times_S T'),
    \qquad [\mathcal L]\longmapsto[g_C^*\mathcal L].
  \]
\end{theorem}
\begin{proof}
  \uses{lem:IsLocallyTrivial_pullback, thm:relative_pic_quotient_well_defined}
  \leanok
  Local triviality of $g_C^*\mathcal L$ follows from
  \cref{lem:IsLocallyTrivial_pullback}.  If
  $e:\mathcal L.\mathrm{carrier}\cong\mathcal L'.\mathrm{carrier}$, the
  functoriality of pullback gives an isomorphism
  $g_C^*\mathcal L.\mathrm{carrier}\cong g_C^*\mathcal L'.\mathrm{carrier}$.
  Hence pullback respects $\sim$ and the universal property of the quotient in
  \cref{thm:relative_pic_quotient_well_defined} gives the displayed map.
\end{proof}
