\chapter{Sheaf cohomology and base change}
\label{pa-ch-cohomology}

\begin{definition}[Module-valued cohomology]
  \label{pa-def-module-cohomology}
  \dcref{III.1,custom}
  \cite{stacks-project}
  For a sheaf \(\mathcal F\) of \(k\)-modules on \(C\), put
  \(H^i(C,\mathcal F)=\operatorname{Ext}^i(k_C,\mathcal F)\), with \(k_C\)
  the constant sheaf.
\end{definition}

\begin{lemma}[Degree zero is sections]
  \label{pa-lem-h0-sections}
  \dcref{III.1,custom}
  \uses{pa-def-module-cohomology}
  \cite{stacks-project}
  \(H^0(C,\mathcal F)\cong\Gamma(C,\mathcal F)\), naturally in \(\mathcal F\).
\end{lemma}

\begin{proof}
  The constant-sheaf adjunction identifies
  \(\operatorname{Hom}(k_C,\mathcal F)\) with evaluation of \(\mathcal F\) on
  the terminal open \(C\).
\end{proof}

\begin{lemma}[Cohomology is functorial]
  \label{pa-lem-cohomology-functorial}
  \dcref{III.1,custom}
  \uses{pa-def-module-cohomology}
  \cite{stacks-project}
  A morphism \(\mathcal F\to\mathcal G\) induces functorial maps
  \(H^i(C,\mathcal F)\to H^i(C,\mathcal G)\).
\end{lemma}

\begin{proof}
  Apply the covariant functoriality of \(\operatorname{Ext}^i(k_C,-)\).
\end{proof}

\begin{proposition}[The Cech short exact sequence]
  \label{pa-prop-cech-short-exact}
  \dcref{III.1,custom}
  \uses{pa-def-cech-complex,pa-lem-affine-vanishing}
  \cite{stacks-project}
  For \(C=U\cup V\) as above,
  \[
    0\to\Gamma(C,\mathcal F)\to\mathcal F(U)\oplus\mathcal F(V)
    \to\mathcal F(U\cap V)\to H^1(C,\mathcal F)\to0
  \]
  is exact.
\end{proposition}

\begin{proof}
  The sheaf equalizer gives exactness at the first three terms, while affine
  acyclicity identifies the cokernel with \(H^1\).
\end{proof}

\begin{definition}[Two-affine Cech complex]
  \label{pa-def-cech-complex}
  \dcref{III.1,custom}
  \cite{stacks-project}
  For \(C=U\cup V\) with \(U,V,U\cap V\) affine, define
  \[
    \check C^0(\mathcal F)=\mathcal F(U)\oplus\mathcal F(V),\qquad
    \check C^1(\mathcal F)=\mathcal F(U\cap V),
  \]
  with \(d(s,t)=s|_{U\cap V}-t|_{U\cap V}\).
\end{definition}

\begin{lemma}[Cech computes \(H^0\) and \(H^1\)]
  \label{pa-lem-cech-computes}
  \dcref{III.1,custom}
  \uses{pa-def-cech-complex}
  \cite{stacks-project}
  For quasi-coherent \(\mathcal F\),
  \(H^0(C,\mathcal F)=\ker d\) and \(H^1(C,\mathcal F)=\operatorname{coker}d\).
\end{lemma}

\begin{proof}
  Quasi-coherent sheaves are acyclic on affine opens and their intersections,
  so the Cech-to-derived spectral sequence for this cover degenerates.
\end{proof}

\begin{lemma}[Affine cohomology vanishes]
  \label{pa-lem-affine-vanishing}
  \dcref{III.1,custom}
  \cite{stacks-project}
  If \(W\) is affine and \(\mathcal F\) is quasi-coherent, then
  \(H^i(W,\mathcal F)=0\) for \(i>0\).
\end{lemma}

\begin{proof}
  For \(W=\Spec A\), write \(\mathcal F=\widetilde M\).  The section functor
  is the exact identity on \(A\)-modules, so its positive derived functors
  vanish.
\end{proof}

\begin{lemma}[Relative curves have cohomological dimension one]
  \label{pa-lem-h2-vanishing}
  \dcref{III.1,custom}
  \uses{pa-lem-cech-computes}
  \cite{stacks-project}
  For every \(T\) and quasi-coherent \(\mathcal F\) on \(C_T\),
  \(H^i(C_T,\mathcal F)=0\) for \(i\ge2\).
\end{lemma}

\begin{proof}
  Base change preserves a finite affine cover with affine finite intersections.
  Its Cech complex has no terms in degrees above one.
\end{proof}

\begin{proposition}[Cech base change]
  \label{pa-prop-cech-base-change}
  \dcref{III.12,custom}
  \uses{pa-def-cech-complex}
  \cite{stacks-project}
  For \(k\subset k'\), the Cech complex of \(\mathcal F_{k'}\) is
  \(\check C^\bullet(\mathcal F)\otimes_k k'\).
\end{proposition}

\begin{proof}
  On affine opens, sections are modules; localization and tensor products
  commute with the flat field extension.  The restriction maps are therefore
  the scalar extensions of \(d\).
\end{proof}

\begin{lemma}[Flat base change on cohomology]
  \label{pa-lem-flat-cohomology}
  \dcref{III.12,custom}
  \uses{pa-prop-cech-base-change,pa-lem-cech-computes}
  \cite{stacks-project}
  The natural map
  \(H^i(C,\mathcal F)\otimes_k k'\to H^i(C_{k'},\mathcal F_{k'})\)
  is an isomorphism.
\end{lemma}

\begin{proof}
  Tensoring with a field is exact, hence commutes with the kernel and cokernel
  of the two-term Cech differential in \ref{pa-lem-cech-computes}.
\end{proof}

\begin{definition}[Dual numbers]
  \label{pa-def-dual-numbers}
  \dcref{5.11,custom}
  \cite{kleiman-picard}
  Put \(k_\epsilon=k[\epsilon]/(\epsilon^2)\) and
  \(C_\epsilon=C\times_k\Spec k_\epsilon\).
\end{definition}

\begin{lemma}[The dual-number unit sequence]
  \label{pa-lem-dual-number-units}
  \dcref{5.11,custom}
  \uses{pa-def-dual-numbers}
  \cite{kleiman-picard}
  There is an exact sequence
  \(1\to\mathcal O_C\to\mathcal O_{C_\epsilon}^{\times}
  \to\mathcal O_C^{\times}\to1\), where \(a\mapsto1+\epsilon a\).
\end{lemma}

\begin{proof}
  Every unit in \(A[\epsilon]/(\epsilon^2)\) is uniquely
  \(u(1+\epsilon a)\), and the latter factors add under multiplication.
  This calculation is compatible with restriction, hence sheafwise exact.
\end{proof}

\begin{theorem}[Finiteness and base change]
  \label{pa-thm-cohomology}
  \dcref{III.12,custom}
  \uses{pa-def-genus}
  \cite{stacks-project}
  If \(\mathcal F\) is coherent on \(C\), then \(H^i(C,\mathcal F)\) is finite
  dimensional over \(k\), and for every extension \(k\subset k'\),
  \[
    H^i(C,\mathcal F)\otimes_k k'\cong
    H^i(C_{k'},\mathcal F_{k'}).
  \]
\end{theorem}

\begin{proof}
  A finite affine cover of the proper curve gives a bounded Cech complex of
  finite-dimensional \(k\)-vector spaces computing coherent cohomology.
  Extending scalars tensors this complex with \(k'\); flatness of a field
  extension commutes with its cohomology.  This proves both finiteness and the
  displayed base-change isomorphism.
\end{proof}

\begin{proposition}[Global functions]
  \label{pa-prop-global-functions}
  \dcref{III.1,custom}
  \cite{hartshorne-ag,stacks-project}
  Geometric integrality and properness imply \(H^0(C,\mathcal O_C)=k\).
\end{proposition}

\begin{proof}
  Apply base change to \(H^0\) and an algebraic closure \(\bar k\).  The
  proper integral scheme \(C_{\bar k}\) has only constant global functions,
  so
  \(H^0(C,\mathcal O_C)\otimes_k\bar k\simeq\bar k\).  The structural
  inclusion of \(k\) is consequently an isomorphism.
\end{proof}

\begin{proposition}[Picard tangent space]
  \label{pa-prop-picard-tangent}
  \dcref{5.11}
  \uses{pa-def-genus,pa-thm-cohomology}
  \cite[Thm.~5.11]{kleiman-picard}
  For the identity element of the relative Picard scheme,
  \[
    T_0\operatorname{Pic}_{C/k}\cong H^1(C,\mathcal O_C),
    \qquad \dim_kT_0\operatorname{Pic}_{C/k}=g(C).
  \]
\end{proposition}

\begin{proof}
  Put \(k[\epsilon]=k[\epsilon]/(\epsilon^2)\).  The kernel of the reduction
  map on the etale Picard functor is computed by
  \[
    1\longrightarrow1+\epsilon\mathcal O_C
    \longrightarrow\mathcal O_{C_{k[\epsilon]}}^\times
    \longrightarrow\mathcal O_C^\times\longrightarrow1.
  \]
  The first term is additively \(\mathcal O_C\), so the kernel is
  \(H^1(C,\mathcal O_C)\).  Scalar substitutions
  \(\epsilon\mapsto a\epsilon\) supply the vector-space structure and the
  preceding base-change theorem gives its compatibility with \(k'\).
\end{proof}
