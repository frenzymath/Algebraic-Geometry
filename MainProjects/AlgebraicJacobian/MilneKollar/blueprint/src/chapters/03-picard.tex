\chapter{The Picard functor}
\label{mk-ch-picard}

\begin{definition}[Relative Picard functor]
  \label{mk-def-picard-functor}
  \dcref{2.2}
  \cite[Def.~2.2]{kleiman-picard}
  For \(T\in(\Sch/k)\), define
  \[
    \operatorname{Pic}^{\sharp}_{C/k}(T)=\operatorname{Pic}(C_T)/
    \operatorname{Pic}(T),
  \]
  where the quotient is by pullback along \(C_T\to T\).  Its associated
  fppf (and etale) sheaf is denoted \(\operatorname{Pic}_{C/k}\).
\end{definition}

\begin{definition}[Relative effective divisors]
  \label{mk-def-relative-divisor}
  \dcref{3.7}
  \cite[Thm.~3.7]{kleiman-picard}
  A relative effective divisor on \(C_T\) is a closed subscheme \(D\subset C_T\)
  that is finite and flat over \(T\).  The functor of such divisors is denoted
  \(\operatorname{Div}_{C/k}\).
\end{definition}

For a smooth relative curve, a finite flat effective Cartier divisor is the
same datum as a relative zero-cycle of degree locally constant on the base.

\begin{proposition}[The Abel map]
  \label{mk-prop-abel-map}
  \dcref{3.7}
  \uses{mk-def-relative-divisor,mk-def-picard-functor}
  \cite[Thm.~3.7]{kleiman-picard}
  The assignment \(D\mapsto\mathcal O_{C_T}(D)\) defines a natural morphism
  \[
    \operatorname{Div}_{C/k}\longrightarrow\operatorname{Pic}_{C/k}.
  \]
  On a geometric fibre it sends an effective divisor to its line-bundle class.
\end{proposition}

\begin{proof}
  An effective Cartier divisor \(D\) determines the invertible sheaf
  \(\mathcal O_{C_T}(D)\).  Pullback of Cartier divisors and of their
  associated line bundles commute with base change in \(T\).  Tensoring by a
  pullback from \(T\) changes the line bundle by the subgroup quotiented out
  in \(\operatorname{Pic}^{\sharp}_{C/k}\), so the construction descends to
  a natural map into the relative Picard sheaf.
\end{proof}

\begin{proposition}[Tensor law]
  \label{mk-prop-picard-group}
  \dcref{2.2,custom}
  \uses{mk-def-picard-functor}
  Tensor product, the trivial line bundle, and duals make
  \(\operatorname{Pic}_{C/k}\) a commutative group-valued functor, naturally
  in \(T\).
\end{proposition}

\begin{proof}
  Tensor product is associative and symmetric up to the canonical
  isomorphisms of line bundles, the trivial bundle is the identity, and
  dualisation supplies inverses.  These operations commute with pullback to
  \(C_T\) and with tensoring by bundles pulled back from \(T\), so they define
  the commutative group structure on the quotient sheaf.
\end{proof}
