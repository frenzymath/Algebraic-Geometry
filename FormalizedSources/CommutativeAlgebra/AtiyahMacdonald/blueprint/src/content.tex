\chapter*{About This Blueprint}
\label{am-about}
This is an independently authored, source-faithful blueprint for M. F. Atiyah
and I. G. Macdonald, \emph{Introduction to Commutative Algebra},
Addison--Wesley, 1969.  It follows the complete mathematical text in source
order.  Definitions, conventions, unnumbered mathematical assertions,
numbered results, and their proofs are retained; examples, exercises,
bibliography, index, and non-mathematical exposition are omitted.  The
printed-number coordinates are preserved by \verb|\dcref| and the visual
PDF-page evidence by \verb|\source|.

\begin{notation}[Notation and terminology]
\label{am-notation}
\source{atiyah-macdonald-commutative-algebra:page-0008}
Rings and modules are denoted by capital italic letters and their elements by
lower-case italic letters; $k$ denotes a field, and $\Z,\Q,\R,\C$ have their
usual meanings.  Maps are written on the left, inclusion uses $\subseteq$ and
strict inclusion $\subsetneq$, and the end of a proof is marked by a square.
The ring convention throughout is commutative with identity, including the
zero ring.
\end{notation}
