\chapter{Rigidity and codimension-one extension}
\label{mk-ch-rigidity}

\begin{theorem}[Rigidity lemma]
  \label{mk-thm-rigidity}
  \dcref{I.2}
  \cite{milne-av,mumford-av}
  Let \(X\) be proper and geometrically connected over \(k\), \(Y\) a reduced
  \(k\)-scheme, and \(f:X\times_kY\to A\) a morphism to an abelian variety.
  If \(f|_{\{x_0\}\times Y}\) is constant, then \(f\) factors uniquely through
  \(X\).  If \(f(x_0,y_0)=0\) and both coordinate restrictions vanish at the
  base points, then
  \[
    f(x,y)=f(x,y_0)+f(x_0,y).
  \]
\end{theorem}

\begin{proof}
  The assertion is unchanged after extending \(k\), so assume it is
  algebraically closed.  Choose an affine neighbourhood \(U\) of the common
  value on \(\{x_0\}\times Y\).  Properness makes the set of \(x\) for which
  \(f(\{x\}\times Y)\) leaves
  \(U\) closed.  On its complement, regular
  functions on \(U\) pull back to functions on \(Y\), and their values are
  fixed by \(x_0\); they are therefore constant.  The equalizer with the
  resulting map \(X\to A\) is closed because \(A\) is separated and contains
  a dense open of the reduced scheme \(X\times Y\), so it is all of
  \(X\times Y\).  Apply this factorisation to
  \(f(x,y)-f(x,y_0)-f(x_0,y)\) to obtain the displayed identity.
\end{proof}

\begin{lemma}[Pointed maps of abelian varieties are homomorphisms]
  \label{mk-lem-pointed-abelian-homomorphism}
  \dcref{I.2}
  \uses{mk-thm-rigidity}
  \cite{milne-av,mumford-av}
  A morphism \(u:A\to B\) of abelian varieties with \(u(0)=0\) is a group
  homomorphism.
\end{lemma}

\begin{proof}
  Apply the decomposition in the rigidity lemma to
  \(F(x,y)=u(x+y)-u(x)-u(y)\).  Both coordinate restrictions are zero, so
  \(F=0\).
\end{proof}

\begin{lemma}[Graph closure of a rational map]
  \label{mk-lem-rational-graph-proper}
  \dcref{3.1}
  \cite{milne-av}
  If \(A\) is proper and \(f:X\dashrightarrow A\) is rational, the closure
  of its graph is proper and birational over \(X\).
\end{lemma}

\begin{proof}
  The graph is closed over the domain of \(f\).  Its closure has proper
  projection to \(X\), being closed in \(X\times A\), and the projection is an
  isomorphism over the dense domain.
\end{proof}

\begin{lemma}[Codimension-one points on a nonsingular variety]
  \label{mk-lem-codim-one-dvr}
  \dcref{3.2}
  \cite{milne-av,hartshorne-ag}
  At a codimension-one point of a nonsingular variety, the local ring is a DVR.
\end{lemma}

\begin{proof}
  Nonsingularity means regularity.  A height-one regular local domain has
  principal maximal ideal and dimension one, hence is a discrete valuation ring.
\end{proof}

\begin{definition}[Rational map domain]
  \label{mk-def-rational-map}
  \dcref{3.1}
  \cite{milne-av}
  A rational map \(X\dashrightarrow Y\) is a class of morphisms from dense
  open subsets of \(X\); its maximal domain is denoted \(\operatorname{Dom}(f)\).
\end{definition}

\begin{lemma}[Generic agreement]
  \label{mk-lem-rational-generic-agreement}
  \dcref{3.1}
  \uses{mk-def-rational-map}
  \cite{milne-av}
  For integral \(X\) and separated \(Y\), rational maps agreeing at the
  generic point agree as rational maps.
\end{lemma}

\begin{proof}
  The equalizer of two representatives is closed and contains the generic
  point, hence contains a dense open of the integral intersection.
\end{proof}

\begin{lemma}[Singleton graph fibres imply extension]
  \label{mk-lem-singleton-graph-extension}
  \dcref{3.2}
  \uses{mk-lem-rational-graph-proper,mk-lem-codim-one-dvr}
  \cite{milne-av}
  If the graph closure of \(f:X\dashrightarrow A\) has one point over every
  \(x\in X\), then it is the graph of a unique morphism \(X\to A\).
\end{lemma}

\begin{proof}
  The graph projection is proper and quasi-finite, hence finite.  Since \(X\)
  is nonsingular, a finite birational map to \(X\) is an isomorphism; compose
  its inverse with the second projection.
\end{proof}

\begin{theorem}[Extension of rational maps]
  \label{mk-thm-rational-extension}
  \dcref{3.1,3.2}
  \uses{mk-thm-rigidity,mk-lem-rational-graph-proper,mk-lem-codim-one-dvr,
    mk-lem-singleton-graph-extension}
  \cite[Thms.~3.1--3.2]{milne-av}
  A rational map from a nonsingular variety to an abelian variety extends to a
  unique morphism.  In particular a rational map from a smooth proper curve to
  an abelian variety has no indeterminacy.
\end{theorem}

\begin{proof}
  Let \(U\) be the domain of definition of the rational map and let
  \(Z\) be its graph closure in \(X\times A\).  Properness of \(A\) makes
  \(Z\to X\) proper and birational.  At a codimension-one point of the
  nonsingular source, the local ring is a discrete valuation ring, so the
  valuative criterion gives a unique point of \(A\) above it.  Thus the
  indeterminacy locus has codimension at least two.  Regular functions on the
  abelian variety are constant along the fibres of \(Z\to X\) by rigidity;
  consequently the graph has only one point over every source point and
  \(Z\to X\) is an isomorphism.  This supplies the unique extension.  For a
  smooth proper curve the same argument is just the valuative criterion at
  its closed points.
\end{proof}
