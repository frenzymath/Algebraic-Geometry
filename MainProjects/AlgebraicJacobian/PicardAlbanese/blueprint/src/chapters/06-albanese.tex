\chapter{The Albanese property}
\label{pa-ch-albanese}

\begin{theorem}[Rigidity]
  \label{pa-thm-rigidity}
  \dcref{I.2}
  \cite{milne-av,mumford-av}
  If \(X\) is proper and geometrically connected, \(A\) is an abelian variety,
  and \(f:X\times_kY\to A\) is constant on \(X\times\{y_0\}\), then \(f\)
  factors through \(Y\).  In particular a pointed morphism from a proper
  connected group variety to \(A\) is a homomorphism.
\end{theorem}

\begin{proof}
  Extend scalars to an algebraic closure and choose an affine neighbourhood
  of the value on \(X\times\{y_0\}\).  Properness makes the projection
  \(X\times Y\to Y\) closed, so the fibres whose image leaves this affine
  neighbourhood form a closed subset of \(Y\).  On its complement, every
  regular function pulled back to a complete connected fibre is constant.
  The value at a base point fixes that constant, so the map factors through
  \(Y\).  Applying the factorisation to the difference of a pointed map and
  its two coordinate restrictions gives the homomorphism assertion.
\end{proof}

\begin{theorem}[Extension of rational maps]
  \label{pa-thm-rational-extension}
  \dcref{3.1,3.2,custom}
  \uses{pa-thm-rigidity}
  \cite[Thms.~3.1--3.2]{milne-av}
  A rational map from a nonsingular variety to an abelian variety extends
  uniquely to a morphism.
\end{theorem}

\begin{proof}
  Take the closure of the graph in \(X\times A\).  Since \(A\) is proper, the
  projection to \(X\) is proper and birational.  At every codimension-one
  point of the nonsingular \(X\), the local ring is a discrete valuation ring,
  and the valuative criterion gives a unique extension there.  The remaining
  indeterminacy has codimension at least two; the rigidity lemma forces every
  fibre of the graph projection to be a singleton.  The graph projection is
  therefore an isomorphism and supplies the unique morphism extending the
  rational map.
\end{proof}

\begin{theorem}[Albanese universal property]
  \label{pa-thm-albanese}
  \dcref{III.6,custom}
  \uses{pa-def-abel-jacobi,pa-thm-pic0-abelian,pa-thm-rigidity}
  For every abelian variety \(A/k\) and every morphism
  \(f:C\to A\) with \(f(P_0)=0\), there is a unique homomorphism
  \(u:\operatorname{Pic}^0_{C/k}\to A\) satisfying
  \[
    f=u\circ\iota_{P_0}.
  \]
\end{theorem}

\begin{proof}
  Choose \(n\ge 2g(C)-1\).  The sum of the values of \(f\) on an ordered
  \(n\)-tuple is symmetric, hence descends to \(f_n:C^{(n)}\to A\).  The
  fibres of \(C^{(n)}\to\operatorname{Pic}^n_{C/k}\) are complete linear
  systems, and the rigidity theorem makes \(f_n\) constant on each fibre.
  It therefore descends to \(\operatorname{Pic}^n_{C/k}\).  Translation by a
  degree-\(n\) line bundle gives a morphism
  \(u:\operatorname{Pic}^0_{C/k}\to A\).  Comparing \(n\) and \(n+1\) shows
  \(u\) respects addition and that \(f=u\circ\iota_{P_0}\).  Surjectivity of
  the symmetric-power map gives uniqueness.
\end{proof}
