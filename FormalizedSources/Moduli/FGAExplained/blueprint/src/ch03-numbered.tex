\begin{remark}[cartesian arrows]\label{fga-rem-3-2}\dcref{3.2}\source{fga-explained:page-0051}\uses{}
The definition of cartesian arrow used here is stronger than the one in SGA~1;
the resulting notions of fibered category nevertheless coincide.
\end{remark}

\begin{remark}[uniqueness of pullbacks]\label{fga-rem-3-3}\dcref{3.3}\source{fga-explained:page-0052}\uses{}
Two pullbacks of an object along the same arrow are uniquely isomorphic: the
unique comparison arrow between them is an isomorphism, with inverse obtained
by exchanging the two pullbacks.
\end{remark}

\begin{proposition}[cartesian arrows]\label{fga-pro-3-4}\dcref{3.4}\source{fga-explained:page-0052}\uses{}
In a category $\mathcal T$ over $\mathcal C$:
\begin{enumerate}
\item composites of cartesian arrows are cartesian;
\item an arrow over an isomorphism of $\mathcal C$ is cartesian if and only if
it is an isomorphism;
\item if $\psi$ is cartesian, then $\varphi$ is cartesian if and only if
$\psi\varphi$ is cartesian;
\item if $F:\mathcal T\to\mathcal G$ and $\mathcal G\to\mathcal C$ are
functors, and both $F(\varphi)$ and the image of $F(\varphi)$ in $\mathcal C$
are cartesian, then $\varphi$ is cartesian over $\mathcal C$.
\end{enumerate}
\end{proposition}
\begin{proof}
Each assertion follows directly by applying the universal property of a
cartesian arrow successively to the relevant factorization problem.
\end{proof}

\begin{definition}[morphism of fibered categories]\label{fga-def-3-6}\dcref{3.6}\source{fga-explained:page-0052}\uses{}
For fibered categories $\mathcal T,\mathcal G$ over $\mathcal C$, a morphism
$F:\mathcal T\to\mathcal G$ is a functor satisfying $p_{\mathcal G}F=p_{\mathcal T}$
and sending cartesian arrows to cartesian arrows.  The equality of structure
functors is literal, not merely up to natural isomorphism.
\end{definition}

\begin{definition}[pseudofunctor]\label{fga-def-3-10}\dcref{3.10}\source{fga-explained:page-0054}\uses{}
A (contravariant) pseudofunctor $\Phi$ on $\mathcal C$ consists of categories
$\Phi(U)$, functors $f^*:\Phi(V)\to\Phi(U)$ for arrows $f:U\to V$,
isomorphisms $\epsilon_U:\operatorname{id}_{\Phi(U)}\Rightarrow(\operatorname{id}_U)^*$,
and isomorphisms $\alpha_{f,g}:f^*g^*\Rightarrow(gf)^*$ for composable arrows.
They satisfy the unit and associativity coherence diagrams.  An ordinary functor
$\mathcal C^{\mathrm{op}}\to\mathbf{Cat}$ is the special case in which all
coherence isomorphisms are identities.
\end{definition}

\begin{remark}[representability by a terminal object]\label{fga-rem-3-27}\dcref{3.27}\source{fga-explained:page-0064}\uses{}
If $F:\mathcal C^{\mathrm{op}}\to\mathbf{Set}$ corresponds to a category
fibered in sets $\mathcal T\to\mathcal C$, then a universal pair for $F$ is a
terminal object of $\mathcal T$.  Thus $F$ is representable exactly when
$\mathcal T$ has a terminal object.  In particular, the representable functor
$h_X(U)=\operatorname{Hom}_{\mathcal C}(U,X)$ corresponds to the comma category
$(\mathcal C/X)$.
\end{remark}

\begin{proposition}[composition of fibered categories]\label{fga-pro-3-28}\dcref{3.28}\source{fga-explained:page-0064}\uses{}
Let $\mathcal T\to\mathcal G\to\mathcal C$ be functors, with $\mathcal T$
fibered over $\mathcal G$.  Then $\mathcal T$ is fibered over $\mathcal C$ if
and only if it is fibered over $\mathcal C$ through the composite.  The same
equivalence holds for being fibered in groupoids or in sets.
\end{proposition}
\begin{proof}
The universal property of a cartesian arrow is unchanged when its base arrow is
viewed in $\mathcal G$ or in $\mathcal C$; the assertions about fibers follow
from the fact that each fiber over $U\in\mathcal C$ is the disjoint union of
the fibers over objects of $\mathcal G$ above $U$.
\end{proof}

\begin{definition}[equivalence of fibered categories]\label{fga-def-3-34}\dcref{3.34}\source{fga-explained:page-0066}\uses{}
An equivalence between fibered categories $\mathcal T$ and $\mathcal G$ over
$\mathcal C$ is a morphism $F:\mathcal T\to\mathcal G$ admitting a morphism
$G:\mathcal G\to\mathcal T$ and isomorphisms $GF\simeq\operatorname{id}_{\mathcal T}$,
$FG\simeq\operatorname{id}_{\mathcal G}$.
\end{definition}

\begin{proposition}[fiberwise equivalence]\label{fga-pro-3-36}\dcref{3.36}\source{fga-explained:page-0066}\uses{}
Let $F:\mathcal T\to\mathcal G$ be a morphism of fibered categories over
$\mathcal C$.  Then $F$ is an equivalence if and only if every induced functor
$F_U:\mathcal T(U)\to\mathcal G(U)$ is an equivalence.
\end{proposition}
\begin{proof}
An inverse restricts fiberwise.  Conversely, fiberwise equivalence gives full
faithfulness by cartesian factorization; choose fiberwise quasi-inverses and
transport arrows along cartesian lifts to obtain a global quasi-inverse.
\end{proof}

\begin{lemma}[fiberwise full faithfulness]\label{fga-lem-3-37}\dcref{3.37}\source{fga-explained:page-0066}\uses{fga-pro-3-36}
If every $F_U$ is fully faithful, then $F$ is fully faithful.
\end{lemma}
\begin{proof}
Given an arrow in $\mathcal G$ over $u:U\to V$, pull back its target along
$u$ by a cartesian arrow.  The given arrow then factors uniquely through an
arrow in the fiber over $U$, which lifts uniquely by the full faithfulness of
$F_U$.
\end{proof}

\begin{proposition}[categories equivalent to sets]\label{fga-pro-3-38}\dcref{3.38}\source{fga-explained:page-0068}\uses{}
A category is equivalent to a set, regarded as a discrete category, if and only
if it is an equivalence relation: every arrow is invertible and between any two
objects there is at most one arrow.
\end{proposition}
\begin{proof}
The property is invariant under equivalence.  Conversely, send each object to
its equivalence class; this functor is fully faithful and essentially surjective.
\end{proof}

\begin{definition}[quasi-functor]\label{fga-def-3-39}\dcref{3.39}\source{fga-explained:page-0068}\uses{fga-def-3-1}
A fibered category $\mathcal F\to\mathcal C$ is a quasi-functor if every fiber
$\mathcal F(U)$ is an equivalence relation.
\end{definition}

\begin{proposition}[quasi-functor criterion]\label{fga-pro-3-40}\dcref{3.40}\source{fga-explained:page-0068}\uses{fga-def-3-39}
A category $\mathcal F$ over $\mathcal C$ is a quasi-functor if and only if:
\begin{enumerate}
\item every arrow $f:U\to p(\eta)$ admits a lift with target $\eta$;
\item for objects $\xi,\eta$ and an arrow $f:p(\xi)\to p(\eta)$, there is at
most one arrow $\xi\to\eta$ over $f$.
\end{enumerate}
\end{proposition}
\begin{proof}
Condition (1) is the existence of cartesian lifts, and condition (2) says that
each fiber is an equivalence relation.  These are precisely the defining
conditions.
\end{proof}

\begin{proposition}[quasi-functors and functors]\label{fga-pro-3-41}\dcref{3.41}\source{fga-explained:page-0068}\uses{fga-pro-3-36,fga-pro-3-38,fga-def-3-39,fga-prop-3-22}
A fibered category over $\mathcal C$ is a quasi-functor if and only if it is
equivalent to a category fibered in sets, hence to a functor
$\mathcal C^{\mathrm{op}}\to\mathbf{Set}$.
\end{proposition}
\begin{proof}
If $\mathcal F$ is equivalent to a functor, each fiber is equivalent to a set.
Conversely, let $\Phi(U)$ be the set of isomorphism classes in $\mathcal F(U)$.
Pullback gives $\Phi:\mathcal C^{\mathrm{op}}\to\mathbf{Set}$, and the
canonical map $\mathcal F\to\Phi$ is a fiberwise equivalence; apply
Proposition~\ref{fga-pro-3-36}.
\end{proof}

\begin{proposition}[morphisms into a fibered category]\label{fga-pro-3-42}\dcref{3.42}\source{fga-explained:page-0068}\uses{fga-def-3-39}
If $\mathcal G$ is fibered in groupoids, then the category of morphisms
$\mathcal F\to\mathcal G$ over $\mathcal C$ is a groupoid.  If $\mathcal G$ is a
quasi-functor, it is an equivalence relation; if $\mathcal G$ is a functor, it
is a set.
\end{proposition}
\begin{proof}
Natural transformations into a groupoid are invertible.  The quasi-functor and
set cases follow by applying the corresponding fiberwise characterization.
\end{proof}

\begin{definition}[representable fibered category]\label{fga-def-3-43}\dcref{3.43}\source{fga-explained:page-0069}\uses{fga-pro-3-41}
A fibered category over $\mathcal C$ is representable if it is equivalent to a
comma category $(\mathcal C/X)$ for some $X\in\mathcal C$.
\end{definition}

\begin{proposition}[representability criterion]\label{fga-pro-3-44}\dcref{3.44}\source{fga-explained:page-0071}\uses{fga-def-3-43}
A fibered category $\mathcal T$ over $\mathcal C$ is representable if and only
if it is fibered in groupoids and there exist $U\in\mathcal C$ and
$\eta\in\mathcal T(U)$ such that every object $\xi$ of $\mathcal T$ admits a
unique arrow $\xi\to\eta$.
\end{proposition}
\begin{proof}
The terminal object $\operatorname{id}_U$ of $(\mathcal C/U)$ has this property.
Conversely, the universal property of $\eta$ identifies $\mathcal T$ with the
comma category $(\mathcal C/U)$, using cartesian lifts to define the comparison
functor.
\end{proof}

\begin{proposition}[equivariant objects]\label{fga-pro-3-49}\dcref{3.49}\source{fga-explained:page-0076}\uses{}
Let $G$ act on an object $X$ and let $\mathcal F$ be a fibered category over
$X$.  A $G$-equivariant structure on $\eta\in\mathcal F(X)$ is an isomorphism
$\varphi:\operatorname{pr}_2^*\eta\xrightarrow{\sim}a^*\eta$ on $G\times X$
whose pullbacks along $G\times G\times X$ satisfy the cocycle condition.
\end{proposition}
\begin{proof}
Choose a cleavage.  The action description in Definition~3.46 gives a
cartesian arrow over $a$; cartesianness identifies it with an isomorphism
$\operatorname{pr}_2^*\eta\simeq a^*\eta$.  The associativity condition for the
group action is exactly the stated cocycle diagram, and the two descriptions
are inverse to one another.
\end{proof}
