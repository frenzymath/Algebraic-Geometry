\section{Double complexes and homological algebra}

\begin{theorem}[First-quadrant double-complex comparison]
\label{mcrt-thm-b-1}\dcref{B.1}
\source{matsumura-commutative-ring-theory:page-0290}\uses{}
Let $K_{p,q}$ be a first-quadrant double complex with horizontal and vertical
differentials.  If $H_q(K_{p,\bullet})=0$ for $q>0$, then the homology of the
total complex is the homology of the complex
$X_p=H_0(K_{p,\bullet})$.  If also $H_p(K_{\bullet,q})=0$ for $p>0$, it is
also the homology of $Y_q=H_0(K_{\bullet,q})$.
\end{theorem}
\begin{proof}
Filter the total complex by the first degree.  The associated spectral sequence
has first page $E^1_{p,q}=H_q(K_{p,\bullet})$, concentrated in the row $q=0$;
it therefore collapses and identifies total homology with $H_\bullet(X)$.
The second assertion follows from the other filtration.
\end{proof}

\begin{theorem}[Injectivity criterion]
\label{mcrt-thm-b-3}\dcref{B.3}
\source{matsumura-commutative-ring-theory:page-0292}\uses{}
An $A$-module $I$ is injective if and only if every $A$-linear map
$\varphi:\mathfrak a\to I$ from an ideal $\mathfrak a\subseteq A$ extends to an
$A$-linear map $A\to I$.
\end{theorem}
\begin{proof}
The forward implication is the extension property applied to the inclusion
$\mathfrak a\hookrightarrow A$.  Conversely, let $N\subseteq M$ and
$f:N\to I$.  Order pairs $(N',f')$ extending $(N,f)$ by inclusion.  A maximal
pair exists by Zorn's lemma.  If $N'\ne M$, choose $x\in M\setminus N'$ and
apply the assumed ideal-extension property to the ideal
$\{a\in A:ax\in N'\}$, defining $a\mapsto f'(ax)$.  This extends $f'$ to
$N'+Ax$, a contradiction.  Hence $N'=M$.
\end{proof}

\begin{theorem}[Essential-extension criterion]
\label{mcrt-thm-b-4}\dcref{B.4}
\source{matsumura-commutative-ring-theory:page-0296}\uses{}
An $A$-module $M$ is injective if and only if every essential extension of
$M$ is equal to $M$.
\end{theorem}
\begin{proof}
If $M$ is injective, any embedding $M\hookrightarrow E$ splits.  Essentiality
forces the complementary summand to be zero.  Conversely, embed $M$ in an
injective module $E$.  The injective envelope of $M$ is an essential extension;
by hypothesis it is $M$, so $M$ is injective.
\end{proof}
