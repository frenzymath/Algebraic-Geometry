\chapter{Representability}
\label{mk-ch-representability}

Fix a relatively ample line bundle on \(C\to\Spec k\).  For a Hilbert
polynomial \(P\), let \(\operatorname{Hilb}^P_{C/k}\) and
\(\operatorname{Quot}^P_{C/k}\) denote the corresponding functors.

\begin{theorem}[Hilbert and Quot representability]
  \label{mk-thm-hilbert-quot}
  \dcref{1.1,custom}
  \cite{nitsure-hilbert-quot}
  The functors \(\operatorname{Hilb}^P_{C/k}\) and
  \(\operatorname{Quot}^P_{C/k}\) are represented by projective \(k\)-schemes.
  Their formation is compatible with base change.
\end{theorem}

\begin{proof}
  Choose a sufficiently positive embedding of \(C\) into projective space.
  A quotient with fixed Hilbert polynomial is generated in one bounded degree,
  so it is represented by a closed subscheme of a Grassmannian.  The
  flattening-stratification condition cuts out the desired Hilbert or Quot
  functor.  The resulting scheme is projective, and the construction commutes
  with arbitrary base change because Hilbert polynomials and flattening are
  preserved by base change.
\end{proof}

\begin{theorem}[Picard representability]
  \label{mk-thm-picard-representability}
  \dcref{4.8}
  \uses{mk-def-picard-functor,mk-def-relative-divisor,mk-thm-hilbert-quot}
  \cite[Thm.~4.8]{kleiman-picard}
  The sheaf \(\operatorname{Pic}_{C/k}\) is represented by a separated
  \(k\)-scheme \(\operatorname{Pic}_{C/k}\), locally of finite type, and
  \[
    \operatorname{Pic}_{C/k}=\coprod_{d\in\mathbb Z}\operatorname{Pic}^d_{C/k}
  \]
  with each degree piece open and quasi-projective.  The universal line bundle
  exists after rigidification along a section (and after etale descent without
  a section).
\end{theorem}

\begin{proof}
  For \(n\) large, Riemann--Roch gives
  \(H^1(C_T,\mathcal L(n))=0\) for every line bundle \(\mathcal L\) in a
  bounded degree range.  Such a line bundle is therefore represented by a
  quotient of a fixed vector bundle of sections, and the Quot scheme gives a
  parameter space.  Two quotients determine the same line bundle precisely
  when they differ by the change of basis of the chosen generators.  The
  resulting fppf equivalence relation is effective; its quotient is the
  corresponding open of the Picard sheaf.  Varying the degree and using
  compatibility on overlaps gives the separated, locally finite-type scheme
  \(\operatorname{Pic}_{C/k}\) and its degree decomposition.
\end{proof}

\begin{lemma}[Quotient presentation]
  \label{mk-lem-picard-quotient}
  \dcref{4.9}
  \uses{mk-thm-hilbert-quot,mk-thm-picard-representability}
  \cite[Lem.~4.9]{kleiman-picard}
  On every bounded degree range, a projective Hilbert/Quot scheme admits an
  fppf locally trivial equivalence relation whose quotient is the
  corresponding open subfunctor of \(\operatorname{Pic}_{C/k}\).
\end{lemma}

\begin{proof}
  On a bounded degree range, the preceding construction uses one fixed
  Quot scheme and a fixed number of generators.  The group of changes of
  generators acts on it, and the orbit relation is fppf locally trivial
  because a line bundle is locally generated by the chosen sections.  Descent
  for line bundles identifies the quotient sheaf with the indicated open
  subfunctor of \(\operatorname{Pic}_{C/k}\).
\end{proof}
