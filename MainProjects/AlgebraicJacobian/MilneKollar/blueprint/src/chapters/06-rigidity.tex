\chapter{Rigidity}
\label{mk-ch-rigidity}

\begin{theorem}[Rigidity lemma]
  \label{mk-thm-rigidity}
  \dcref{I.2}
  \cite{milne-av,mumford-av}
  Let \(X\) be proper and geometrically connected over \(k\), \(Y\) a
  \(k\)-scheme, and \(f:X\times_kY\to A\) a morphism to an abelian variety.
  If \(f|_{X\times\{y_0\}}\) is constant, then \(f\) factors uniquely through
  \(X\times Y\to Y\).  If \(f(x_0,y_0)=0\) and both coordinate restrictions
  vanish at the base points, then
  \[
    f(x,y)=f(x,y_0)+f(x_0,y).
  \]
\end{theorem}

\begin{theorem}[Extension of rational maps]
  \label{mk-thm-rational-extension}
  \dcref{3.1,3.2,custom}
  \uses{mk-thm-rigidity}
  \cite[Thms.~3.1--3.2]{milne-av}
  A rational map from a nonsingular variety to an abelian variety extends to a
  unique morphism.  In particular a rational map from a smooth proper curve to
  an abelian variety has no indeterminacy.
\end{theorem}
