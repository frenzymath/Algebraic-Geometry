\chapter*{Introductory mathematical assertions}
\begin{remark}[Picard groups and the Picard scheme]\label{kps-intro-assertions}
\source{kleiman-picard:page-0001}
For a ringed space $X$, isomorphism classes of invertible sheaves form the
group $\Pic(X)$. If $X$ is an integral projective scheme over an algebraically
closed field, these classes are represented by a natural $k$-scheme
$\IPic_{X/k}$, a disjoint union of quasi-projective schemes; tensor product and
duals induce its group law and inverse. In characteristic zero it is reduced,
whereas in positive characteristic it need not be reduced, and the construction
fits into algebraic families.
\end{remark}
\begin{remark}[Abel's addition theorem]
\label{kps-intro-abel-theorem}
\source{kleiman-picard:page-0003}
For an algebraic function $y=y(x)$ defined by
$f(x,y)=y^n+f_1(x)y^{n-1}+\cdots+f_n(x)=0$, an Abelian integral is
$\psi_x=\int_{x_0}^xR(x,y)\,dx$. If $p$ is the genus, Abel's addition theorem
expresses a suitable sum of such integrals in terms of $p$ new points and an
elementary term, and $p$ is minimal for this property. In the hyperelliptic
case $y^2=\varphi(x)$ with $\varphi$ square-free of degree $d$,
\[
  p=\begin{cases}(d-1)/2,&d\text{ odd},\\(d-2)/2,&d\text{ even}.\end{cases}
\]
\end{remark}

\begin{remark}[Abelian integrals and genus]
\label{kps-intro-abelian-integrals}
\source{kleiman-picard:page-0005}
For a plane curve $f(x,y)=0$, Abelian integrals have the form
\[
  \psi_x=\int_{x_0}^x \frac{h(x,y)}{\partial f/\partial y}\,dx,
  \qquad \deg h\leq \deg f-3.
\]
The independent numerators are counted by the genus $p$. If a variable
effective divisor has degree $\mu$ and its complete linear system has
dimension $\alpha$, then Abel's relations give
\[
  \mu-\alpha\geq p,\qquad \mu-\alpha=p-i\quad(i\geq 0),
\]
and for a plane curve of degree $d$ with $\delta$ nodes and $\kappa$ cusps,
$p=(d-1)(d-2)/2-\delta-\kappa$.
\end{remark}

\begin{remark}[Abel maps and Jacobians]
\label{kps-intro-jacobian}
\source{kleiman-picard:page-0006}
Choosing $p$ first-kind integrals and $2p$ homology periods gives a lattice
$L\subset\mathbb C^p$ and a complex torus $J=\mathbb C^p/L$. For every
$\mu$, the Abel map
\[
  \Psi_\mu:C^{(\mu)}\longrightarrow J,
  \qquad (x_1,\ldots,x_\mu)\longmapsto
  \left(\sum_i\psi_1(x_i),\ldots,\sum_i\psi_p(x_i)\right)
\]
has fibers equal to complete linear systems: two divisors are linearly
equivalent exactly when they have the same image. For $\mu\geq p$, the
Jacobian is the quotient of $C^{(\mu)}$ by linear equivalence, and its
addition and Abel map are algebraic.
\end{remark}

\begin{remark}[Surface invariants and Picard dimension]
\label{kps-intro-surface-invariants}
\source{kleiman-picard:page-0009}
For a surface, the geometric and arithmetic genera $p_g$ and $p_a$ measure
the irregularity; the Picard variety has dimension
\[
  \dim P=p_g-p_a=q=b/2,
\]
where $q$ is the third-kind rank and $b$ is the second Betti number. Curves
are thereby organized both by linear equivalence and by the
Neron--Severi group modulo algebraic equivalence.
\end{remark}

\begin{remark}[Picard and Albanese constructions]
\label{kps-intro-picard-construction}
\source{kleiman-picard:page-0013}
For a complete normal variety $X$, the Picard variety $P$ parametrizes
linear-equivalence classes of divisors and the Albanese variety $A$ receives
the universal map from $X$; $P$ and $A$ are dual Abelian varieties (and are
the Jacobian and Abel map for a curve). Grothendieck's Hilbert and Picard
schemes impose flatness as the continuity condition in families: one defines
functors of points, sheafifies the naive Picard functor in the
Zariski/etale/fppf topologies, and covers the resulting functor by
representable open subfunctors of Hilbert schemes, with quotients supplying
the Picard scheme.
\end{remark}
