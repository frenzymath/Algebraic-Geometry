\chapter{Rigidity}
\label{mk-ch-rigidity}

\begin{theorem}[Rigidity lemma]
  \label{mk-thm-rigidity}
  \dcref{I.2}
  \cite{milne-av,mumford-av}
  Let \(X\) be proper and geometrically connected over \(k\), \(Y\) a
  \(k\)-scheme, and \(f:X\times_kY\to A\) a morphism to an abelian variety.
  If \(f|_{X\times\{y_0\}}\) is constant, then \(f\) factors uniquely through
  \(X\times Y\to Y\).  If \(f(x_0,y_0)=0\) and both coordinate restrictions
  vanish at the base points, then
  \[
    f(x,y)=f(x,y_0)+f(x_0,y).
  \]
\end{theorem}

\begin{proof}
  The assertion is unchanged after extending \(k\), so assume it is
  algebraically closed.  Let \(U\) be an affine neighbourhood of the target
  value \(f(x_0,y_0)\).  Properness of \(X\) makes the projection
  \(X\times Y\to Y\) closed, so the set of \(y\) for which
  \(f(X\times\{y\})\) is not contained in \(U\) is closed.  Its complement is
  a nonempty open neighbourhood of \(y_0\).  On each fibre over this open,
  every regular function pulled back from \(U\) is constant because \(X\) is
  complete and connected.  The value is fixed by the section \(x_0\), hence
  \(f\) is constant on a dense open subset.  Since the target is separated,
  the equalizer with the constant map is closed and therefore all of
  \(X\times Y\).  Applying this to the difference
  \(f(x,y)-f(x,y_0)-f(x_0,y)\) gives the final identity.
\end{proof}

\begin{theorem}[Extension of rational maps]
  \label{mk-thm-rational-extension}
  \dcref{3.1,3.2,custom}
  \uses{mk-thm-rigidity}
  \cite[Thms.~3.1--3.2]{milne-av}
  A rational map from a nonsingular variety to an abelian variety extends to a
  unique morphism.  In particular a rational map from a smooth proper curve to
  an abelian variety has no indeterminacy.
\end{theorem}

\begin{proof}
  Let \(U\) be the domain of definition of the rational map and let
  \(Z\) be its graph closure in \(X\times A\).  Properness of \(A\) makes
  \(Z\to X\) proper and birational.  At a codimension-one point of the
  nonsingular source, the local ring is a discrete valuation ring, so the
  valuative criterion gives a unique point of \(A\) above it.  Thus the
  indeterminacy locus has codimension at least two.  Regular functions on the
  abelian variety are constant along the fibres of \(Z\to X\) by rigidity;
  consequently the graph has only one point over every source point and
  \(Z\to X\) is an isomorphism.  This supplies the unique extension.  For a
  smooth proper curve the same argument is just the valuative criterion at
  its closed points.
\end{proof}
