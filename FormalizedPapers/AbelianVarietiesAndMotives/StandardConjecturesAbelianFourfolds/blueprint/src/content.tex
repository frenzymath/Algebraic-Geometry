\chapter*{About This Blueprint}
\label{standard-conjectures-abelian-fourfolds-about}

\textbf{Edition and PDF.} ArXiv preprint \href{https://arxiv.org/abs/1806.03216}{arXiv:1806.03216};
\href{https://arxiv.org/pdf/1806.03216}{direct PDF}.
\textbf{Summary.} The paper proves the standard conjecture of Hodge type for abelian fourfolds using
$p$-adic Hodge theory, and derives consequences for numerical and $\ell$-adic homological equivalence.
\textbf{Reading warning.} This blueprint rephrases and distills the original paper; read it alongside the original.
