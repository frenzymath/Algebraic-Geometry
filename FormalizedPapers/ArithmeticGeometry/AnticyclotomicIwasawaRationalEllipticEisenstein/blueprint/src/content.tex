\chapter*{About This Blueprint}
\label{anticyclotomic-iwasawa-rational-elliptic-eisenstein-about}

\textbf{Edition and PDF.} ArXiv preprint \href{https://arxiv.org/abs/2008.02571}{arXiv:2008.02571};
\href{https://arxiv.org/pdf/2008.02571}{direct PDF}.
\textbf{Summary.} The paper relates anticyclotomic Iwasawa theory for rational elliptic curves at split Eisenstein primes
to character theory, proving Heegner-point, $p$-converse, and analytic-rank-one BSD results.
\textbf{Reading warning.} This blueprint rephrases and distills the original paper; read it alongside the original.
