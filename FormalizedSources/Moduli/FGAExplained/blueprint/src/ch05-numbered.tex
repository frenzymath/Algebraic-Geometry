\begin{theorem}[linear scheme representing relative Hom]\label{fga-the-5-8}\dcref{5.8}\source{fga-explained:page-0130}\uses{fga-def-5-1,fga-thm-5-7}
Let $S$ be a noetherian scheme and $\pi:X\to S$ a projective morphism.  Let
$\mathcal E$ and $\mathcal F$ be coherent sheaves on $X$, with $\mathcal F$
flat over $S$.  The contravariant functor on $S$-schemes
\[
T\longmapsto
\operatorname{Hom}_{\mathcal O_{X_T}}(\mathcal E_T,\mathcal F_T)
\]
is represented by a linear scheme over $S$.
\end{theorem}
\begin{proof}
For $\mathcal E$ locally free, cohomology and base change identify the functor
with the functor of sections of a coherent sheaf $\mathcal Q$ on $S$:
\[
\operatorname{Hom}_{X_T}(\mathcal E_T,\mathcal F_T)
\simeq\operatorname{Hom}_{S}(\mathcal Q,\mathcal O_T).
\]
It is therefore represented by $\operatorname{Spec}_S\operatorname{Sym}\mathcal Q$.
Locally on $S$, choose a right-exact presentation
$\mathcal E_1\to\mathcal E_0\to\mathcal E\to0$ by locally free sheaves.
The cokernel of the induced map between the corresponding sheaves $\mathcal Q_1$
and $\mathcal Q_0$ represents the Hom functor for $\mathcal E$; the local
linear schemes glue, giving the required representing scheme.
\end{proof}

\begin{remark}[zero section of the linear scheme]\label{fga-rem-5-9}\dcref{5.9}\source{fga-explained:page-0131}\uses{fga-the-5-8}
The zero section $V_0\subset V$ is the locus where the universal homomorphism
vanishes.  If $f:T\to V$ corresponds to a homomorphism
$\mathcal E_T\to\mathcal F_T$, then $f^{-1}(V_0)$ is the closed subscheme with
the universal property that a morphism $U\to T$ factors through $f^{-1}(V_0)$
if and only if the pullback of the homomorphism to $X_U$ is zero.
\end{remark}
