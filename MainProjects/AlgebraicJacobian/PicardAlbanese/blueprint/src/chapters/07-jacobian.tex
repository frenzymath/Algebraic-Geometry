\chapter{The Jacobian}
\label{pa-ch-jacobian}

\begin{definition}[Jacobian]
  \label{pa-def-jacobian}
  \dcref{III.6,custom}
  \uses{pa-def-pic0}
  \[
    \operatorname{Jac}(C):=\operatorname{Pic}^0_{C/k}.
  \]
\end{definition}

\begin{theorem}[Jacobian theorem]
  \label{pa-thm-jacobian}
  \dcref{III.6,5.13,5.14,custom}
  \uses{pa-def-jacobian,pa-thm-pic0-abelian,pa-thm-albanese}
  By \ref{pa-thm-pic0-abelian}, the Jacobian is an abelian variety of dimension
  \(g(C)\), and by \ref{pa-thm-albanese}, \(\iota_{P_0}\) is universal among
  pointed maps from \(C\) to abelian varieties.  Changing \(P_0\) changes
  \(\iota_{P_0}\) by a translation.
\end{theorem}

\begin{theorem}[Base change]
  \label{pa-thm-jacobian-base-change}
  \dcref{III.6,custom}
  \uses{pa-def-jacobian,pa-thm-picard-representability}
  For every field extension \(k\subset k'\), there is a canonical isomorphism
  \[
    \operatorname{Jac}(C)\times_k k'\cong
    \operatorname{Jac}(C\times_k k'),
  \]
  compatible with towers and with the Abel--Jacobi maps.
\end{theorem}
