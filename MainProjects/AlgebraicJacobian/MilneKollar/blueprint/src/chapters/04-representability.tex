Fix a relatively ample line bundle on \(C\to\Spec k\).  For a Hilbert
polynomial \(P\), let \(\operatorname{Hilb}^P_{C/k}\) and
\(\operatorname{Quot}^P_{C/k}\) denote the corresponding functors.

\begin{definition}[Section module in a fixed twist]
  \label{mk-def-section-module}
  \dcref{4.8,custom}
  \cite{kleiman-picard}
  For a line bundle \(\mathcal L\) and twist \(m\), write
  \(M_m(\mathcal L)=H^0(C,\mathcal L(m))\).
\end{definition}

\begin{lemma}[Finite generation of the section module]
  \label{mk-lem-section-module-finite}
  \dcref{4.8,custom}
  \uses{mk-def-section-module}
  \cite{hartshorne-ag,kleiman-picard}
  \(M_m(\mathcal L)\) is finite-dimensional over \(k\), and in a flat family
  its formation is compatible with base change whenever
  \(H^1(C,\mathcal L(m))=0\).
\end{lemma}

\begin{proof}
  Coherent-cohomology finiteness gives finite dimension.  Cohomology and base
  change identifies sections after a field extension; vanishing of \(H^1\)
  removes the cokernel in the base-change exact sequence.
\end{proof}

\begin{lemma}[Effective divisors are flat length families]
  \label{mk-lem-effective-divisor-flat-length}
  \dcref{3.7,custom}
  \cite{kleiman-picard}
  A relative effective divisor \(D\subset C_T\) is a finite locally free
  \(T\)-scheme, and its rank is its fibrewise degree.
\end{lemma}

\begin{proof}
  A relative Cartier divisor on a smooth relative curve has
  \(\mathcal O_D\) finite over \(\mathcal O_T\).  Flatness over \(T\) makes
  this finite module locally free, with rank equal to its geometric fibre
  length.
\end{proof}

\begin{proposition}[The divisor functor is a Hilbert open]
  \label{mk-prop-divisor-hilbert-open}
  \dcref{3.7,custom}
  \uses{mk-lem-effective-divisor-flat-length,mk-thm-hilbert-quot}
  \cite{kleiman-picard,nitsure-hilbert-quot}
  The functor of degree-\(n\) relative effective divisors is represented by
  the open-and-closed length-\(n\) locus of \(\operatorname{Hilb}_{C/k}\).
\end{proposition}

\begin{proof}
  By \ref{mk-lem-effective-divisor-flat-length}, such a divisor is a flat
  family with constant Hilbert polynomial \(n\).  On a smooth curve every
  finite flat length-\(n\) subscheme is an effective Cartier divisor, so the
  two functors agree.  The Hilbert scheme theorem gives representability.
\end{proof}

\begin{lemma}[Large twists are generated and acyclic]
  \label{mk-lem-large-twist}
  \dcref{4.8,custom}
  \uses{mk-thm-riemann-roch}
  \cite{hartshorne-ag,kleiman-picard}
  For a bounded set of degrees there is \(m\) such that every line bundle
  \(\mathcal L\) in that set satisfies
  \(H^1(C,\mathcal L(m))=0\) and is generated by its global sections.
\end{lemma}

\begin{proof}
  The first assertion is the large-degree vanishing
  \ref{mk-cor-large-degree}.  For a closed point \(P\), apply that vanishing
  to \(\mathcal L(m-P)\); the evaluation map at \(P\) is then surjective.
  Since this holds for every \(P\), the bundle is globally generated.
\end{proof}

\begin{proposition}[Quot parameterises generated line bundles]
  \label{mk-prop-quot-line-bundles}
  \dcref{4.8,custom}
  \uses{mk-lem-large-twist,mk-thm-hilbert-quot}
  \cite{kleiman-picard,nitsure-hilbert-quot}
  With \(m\) as above, line bundles \(\mathcal L\) in the bounded range are
  represented by quotients
  \(H^0(C,\mathcal L(m))\otimes\mathcal O_C(-m)\twoheadrightarrow\mathcal L\)
  in a fixed Quot scheme.
\end{proposition}

\begin{proof}
  Global generation gives the evaluation quotient.  The Hilbert polynomial is
  fixed on the bounded degree range, so the quotient lies in one projective
  Quot component.  Conversely, the open locus where the quotient is invertible
  gives a line bundle with the prescribed polynomial.
\end{proof}

\begin{lemma}[Change of generators is the equivalence relation]
  \label{mk-lem-quot-generator-relation}
  \dcref{4.9,custom}
  \uses{mk-prop-quot-line-bundles}
  \cite{kleiman-picard}
  Two Quot points give the same relative line-bundle class exactly when,
  fppf-locally on the base, they differ by an element of the corresponding
  general linear group.
\end{lemma}

\begin{proof}
  An isomorphism of the quotient line bundles carries one generating basis to
  the other, giving a change-of-generators matrix.  Conversely such a matrix
  identifies the two evaluation quotients.  Local freeness of the generating
  module makes this relation fppf-locally defined.
\end{proof}

\begin{proposition}[Quotient descent on a bounded degree range]
  \label{mk-prop-picard-quotient-descent}
  \dcref{4.9,custom}
  \uses{mk-lem-quot-generator-relation,mk-prop-picard-sheaf}
  \cite{kleiman-picard}
  The fppf quotient of the invertible Quot locus by change of generators
  represents the corresponding open subfunctor of \(\operatorname{Pic}_{C/k}\).
\end{proposition}

\begin{proof}
  The relation is an effective fppf equivalence relation by descent for
  line bundles.  Its sheaf quotient has exactly the same \(T\)-points as
  relative line-bundle classes in the bounded range, and the quotient theorem
  supplies the representing scheme.
\end{proof}

\begin{theorem}[Hilbert and Quot representability]
  \label{mk-thm-hilbert-quot}
  \dcref{1.1,custom}
  \cite{nitsure-hilbert-quot}
  The functors \(\operatorname{Hilb}^P_{C/k}\) and
  \(\operatorname{Quot}^P_{C/k}\) are represented by projective \(k\)-schemes.
  Their formation is compatible with base change.
\end{theorem}

\begin{proof}
  Choose a sufficiently positive embedding of \(C\) into projective space.
  A quotient with fixed Hilbert polynomial is generated in one bounded degree,
  so it is represented by a closed subscheme of a Grassmannian.  The
  flattening-stratification condition cuts out the desired Hilbert or Quot
  functor.  The resulting scheme is projective, and the construction commutes
  with arbitrary base change because Hilbert polynomials and flattening are
  preserved by base change.
\end{proof}

\begin{lemma}[Flattening stratification]
  \label{mk-lem-flattening-stratification}
  \dcref{1.1,custom}
  \cite{nitsure-hilbert-quot}
  For a projective morphism and a coherent sheaf, there is a universal
  locally closed stratification of the base on which the sheaf is flat with a
  fixed Hilbert polynomial.
\end{lemma}

\begin{proof}
  Choose a finite graded presentation after a projective embedding.  Fitting
  ideals of the graded pieces cut out the loci with the prescribed ranks;
  these loci glue and satisfy the universal factorisation property.
\end{proof}

\begin{lemma}[Invertibility is open in a flat family]
  \label{mk-lem-invertibility-open}
  \dcref{4.8,custom}
  \uses{mk-lem-flattening-stratification}
  \cite{kleiman-picard}
  In a flat family of rank-one quotients on a smooth curve, the locus where
  the quotient is invertible is open.
\end{lemma}

\begin{proof}
  The locally free locus of a flat coherent sheaf is open.  Rank one on the
  generic point and the fixed Hilbert polynomial make this the invertible locus.
\end{proof}

\begin{theorem}[Picard representability]
  \label{mk-thm-picard-representability}
  \dcref{4.8}
  \uses{mk-def-picard-functor,mk-def-relative-divisor,mk-thm-hilbert-quot}
  \cite[Thm.~4.8]{kleiman-picard}
  The sheaf \(\operatorname{Pic}_{C/k}\) is represented by a separated
  \(k\)-scheme \(\operatorname{Pic}_{C/k}\), locally of finite type, and
  \[
    \operatorname{Pic}_{C/k}=\coprod_{d\in\mathbb Z}\operatorname{Pic}^d_{C/k}
  \]
  with each degree piece open and quasi-projective.  The universal line bundle
  exists after rigidification along a section (and after etale descent without
  a section).
\end{theorem}

\begin{proof}
  For \(n\) large, Riemann--Roch gives
  \(H^1(C_T,\mathcal L(n))=0\) for every line bundle \(\mathcal L\) in a
  bounded degree range.  Such a line bundle is therefore represented by a
  quotient of a fixed vector bundle of sections, and the Quot scheme gives a
  parameter space.  Two quotients determine the same line bundle precisely
  when they differ by the change of basis of the chosen generators.  The
  resulting fppf equivalence relation is effective; its quotient is the
  corresponding open of the Picard sheaf.  Varying the degree and using
  compatibility on overlaps gives the separated, locally finite-type scheme
  \(\operatorname{Pic}_{C/k}\) and its degree decomposition.
\end{proof}

\begin{lemma}[Quotient presentation]
  \label{mk-lem-picard-quotient}
  \dcref{4.9}
  \uses{mk-thm-hilbert-quot,mk-thm-picard-representability}
  \cite[Lem.~4.9]{kleiman-picard}
  On every bounded degree range, a projective Hilbert/Quot scheme admits an
  fppf locally trivial equivalence relation whose quotient is the
  corresponding open subfunctor of \(\operatorname{Pic}_{C/k}\).
\end{lemma}

\begin{proof}
  On a bounded degree range, the preceding construction uses one fixed
  Quot scheme and a fixed number of generators.  The group of changes of
  generators acts on it, and the orbit relation is fppf locally trivial
  because a line bundle is locally generated by the chosen sections.  Descent
  for line bundles identifies the quotient sheaf with the indicated open
  subfunctor of \(\operatorname{Pic}_{C/k}\).
\end{proof}
