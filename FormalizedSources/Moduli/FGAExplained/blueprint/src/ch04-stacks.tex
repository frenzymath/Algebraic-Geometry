\begin{definition}[descent datum]\label{fga-def-4-2}\dcref{4.2}\source{fga-explained:page-0080}\uses{fga-def-3-1}
For a covering $\{U_i\to U\}$ and $\xi_i\in\mathcal X(U_i)$, a descent datum
is a family of isomorphisms
$\phi_{ij}:\operatorname{pr}_1^*\xi_i\simeq\operatorname{pr}_2^*\xi_j$
on $U_i\times_UU_j$ satisfying $\phi_{ii}=1$ and the cocycle condition on
$U_i\times_UU_j\times_UU_k$.
\end{definition}
\begin{definition}[effective descent datum]\label{fga-def-4-6}\dcref{4.6}\source{fga-explained:page-0082}\uses{fga-def-4-2}
A descent datum is effective if it is isomorphic to the canonical datum obtained
by pulling back an object $\xi\in\mathcal X(U)$, and the isomorphism is unique
up to a unique vertical isomorphism.
\end{definition}
\begin{proposition}\label{fga-prop-4-1}\dcref{4.1}\source{fga-explained:page-0079}\uses{fga-def-4-2}
Descent data on a cover form a category $\mathcal X_{\mathrm{desc}}(U)$, and
pullback from $\mathcal X(U)$ defines a functor
$\mathcal X(U)\to\mathcal X_{\mathrm{desc}}(U)$.
\end{proposition}
\begin{proposition}\label{fga-prop-4-5}\dcref{4.5}\source{fga-explained:page-0082}\uses{fga-def-4-6,fga-thm-2-55}
For a representable fibered category $h_X$, the functor from $h_X(U)$ to the
category of descent data is an equivalence.
\end{proposition}

\begin{definition}[prestack and stack]\label{fga-def-4-8}\dcref{4.8}\source{fga-explained:page-0085}\uses{fga-def-4-6}
A fibered category over a site is a prestack if morphisms satisfy effective
descent (the presheaf of arrows is a sheaf).  It is a stack if, in addition,
every descent datum for objects is effective.
\end{definition}
\begin{proposition}\label{fga-prop-4-7}\dcref{4.7}\source{fga-explained:page-0084}\uses{fga-prop-4-1}
An object descent functor is an equivalence exactly when it is fully faithful on
morphisms and essentially surjective on objects.
\end{proposition}
\begin{proposition}\label{fga-prop-4-9}\dcref{4.9}\source{fga-explained:page-0086}\uses{fga-def-2-37,fga-def-4-2}
For a set-valued functor, descent data reduce to the equalizer diagram defining
the sheaf condition.
\end{proposition}
\begin{proposition}\label{fga-prop-4-12}\dcref{4.12}\source{fga-explained:page-0088}\uses{fga-def-4-8}
The prestack condition can be checked after refining a covering, and the stack
condition is local for the topology.
\end{proposition}
\begin{corollary}\label{fga-cor-4-13}\dcref{4.13}\source{fga-explained:page-0088}\uses{fga-prop-4-7,fga-prop-4-12}
A fibered category is a stack if and only if for every covering its descent
functor is an equivalence.
\end{corollary}
\begin{proposition}\label{fga-prop-4-14}\dcref{4.14}\source{fga-explained:page-0089}\uses{fga-cor-4-13}
It is enough to test stack descent on a basis of covering families, or on one
cover after arbitrary base change.
\end{proposition}
\begin{lemma}\label{fga-lem-4-15}\dcref{4.15}\source{fga-explained:page-0090}\uses{fga-def-4-8,fga-def-2-41}
For a prestack, descent data with respect to two sieves $S\subseteq S'$ agree
whenever $S$ is covering; restriction is fully faithful.
\end{lemma}
\begin{proposition}\label{fga-prop-4-16}\dcref{4.16}\source{fga-explained:page-0090}\uses{fga-lem-4-15}
If $T\subseteq T'$ are topologies, a $T'$-stack is a $T$-stack; if every
$T'$-cover refines a $T$-cover, the converse holds.
\end{proposition}
\begin{lemma}\label{fga-lem-4-17}\dcref{4.17}\source{fga-explained:page-0091}\uses{fga-lem-4-15}
For a prestack, descent along a cover is unchanged by replacing the cover with
a common refinement or by pulling it back along another cover.
\end{lemma}
\begin{definition}[substack]\label{fga-def-4-18}\dcref{4.18}\source{fga-explained:page-0091}\uses{fga-def-4-8}
A substack is a full sub-fibered category closed under isomorphism and effective
descent.
\end{definition}

\section{Descent for quasi-coherent sheaves}
\source{fga-explained:page-0089}\source{fga-explained:page-0090}\source{fga-explained:page-0091}\source{fga-explained:page-0092}\source{fga-explained:page-0093}\source{fga-explained:page-0094}\source{fga-explained:page-0095}\source{fga-explained:page-0096}\source{fga-explained:page-0097}\source{fga-explained:page-0098}

\begin{theorem}[fpqc descent for quasi-coherent sheaves]\label{fga-thm-4-23}\dcref{4.23}\source{fga-explained:page-0119}\uses{fga-def-4-8,fga-def-2-34}
For a scheme $S$, the fibered category $(\operatorname{QCoh}/S)$ of
quasi-coherent sheaves is a stack over $(\operatorname{Sch}/S)$ for the fpqc
topology.
\end{theorem}
\begin{proof}
Morphisms of quasi-coherent sheaves form fpqc sheaves by the module equalizer
criterion.  On affine pieces a descent datum is a module with compatible
semilinear identifications over $B\otimes_A B$; faithful flatness makes the
equalizer an $A$-module whose base change recovers the datum.  Compatibility on
overlaps glues the affine modules and their restrictions, and quasi-coherence
is local.
\end{proof}
\begin{theorem}\label{fga-thm-4-21}\dcref{4.21}\source{fga-explained:page-0118}\uses{fga-thm-4-23}
For an fpqc cover $T\to S$, the category of quasi-coherent sheaves on $S$ is
equivalent to the category of quasi-coherent sheaves on $T$ equipped with
effective descent data.
\end{theorem}
\begin{corollary}\label{fga-cor-4-34}\dcref{4.34}\source{fga-explained:page-0130}\uses{fga-thm-4-23}
Properties of quasi-coherent modules defined by local module conditions descend
along fpqc covers; in particular finite type, finite presentation, and local
freeness descend.
\end{corollary}
\begin{lemma}\label{fga-lem-4-30}\dcref{4.30}\source{fga-explained:page-0129}\uses{fga-thm-4-21}
If $\{U_i\to U\}$ is fpqc and modules $M_i$ carry descent data satisfying the
cocycle condition, the equalizer module on affine charts is quasi-coherent and
base changes back to each $M_i$.
\end{lemma}

\section{Descent for morphisms of schemes}
\source{fga-explained:page-0099}\source{fga-explained:page-0100}\source{fga-explained:page-0101}\source{fga-explained:page-0102}\source{fga-explained:page-0103}\source{fga-explained:page-0104}\source{fga-explained:page-0105}\source{fga-explained:page-0106}\source{fga-explained:page-0107}\source{fga-explained:page-0108}

\begin{proposition}\label{fga-prop-4-31}\dcref{4.31}\source{fga-explained:page-0133}\uses{fga-def-3-16,fga-def-4-8}
For a subcanonical site and a stable class of morphisms, the fibered category of
objects whose structure map lies in the class is a prestack.
\end{proposition}
\begin{definition}[local class of arrows]\label{fga-def-4-32}\dcref{4.32}\source{fga-explained:page-0134}\uses{fga-def-3-16}
A stable class of arrows is local if membership can be checked after a covering
base change.
\end{definition}
\begin{theorem}\label{fga-thm-4-33}\dcref{4.33}\source{fga-explained:page-0135}\uses{fga-thm-4-23,fga-prop-4-31}
The fibered category $(\operatorname{Aff}/S)$ of affine schemes over $S$ is a
stack for the fpqc topology.
\end{theorem}
\begin{corollary}\label{fga-cor-4-36}\dcref{4.36}\source{fga-explained:page-0136}\uses{fga-thm-4-33}
Affineness and the usual finiteness properties of affine morphisms descend
along fpqc covers.
\end{corollary}
\begin{theorem}\label{fga-thm-4-38}\dcref{4.38}\source{fga-explained:page-0138}\uses{fga-thm-4-23,fga-def-4-8}
Flat proper morphisms with a relatively ample line bundle, and in particular
projective morphisms, satisfy effective fpqc descent.
\end{theorem}
\begin{proposition}\label{fga-prop-4-40}\dcref{4.40}\source{fga-explained:page-0140}\uses{fga-thm-4-23}
Descent data for a relatively ample invertible sheaf descend after replacing it
by a sufficiently high tensor power; the induced morphism to projective space
then descends.
\end{proposition}

\section{Descent along torsors}
\source{fga-explained:page-0109}\source{fga-explained:page-0110}\source{fga-explained:page-0111}\source{fga-explained:page-0112}\source{fga-explained:page-0113}\source{fga-explained:page-0114}

\begin{definition}[torsor]\label{fga-def-4-42}\dcref{4.42}\source{fga-explained:page-0099}\uses{fga-def-2-15}
For a group object $G$, a $G$-torsor is an object $X$ with a right $G$-action
such that $X\to U$ is a covering and $X\times G\to X\times_U X$,
$(x,g)\mapsto(x,xg)$, is an isomorphism.
\end{definition}
\begin{proposition}\label{fga-prop-4-43}\dcref{4.43}\source{fga-explained:page-0100}\uses{fga-def-4-42,fga-def-4-2}
For a torsor, equivariant objects on the total space are precisely descent data
for the covering $X\to U$.
\end{proposition}
\begin{lemma}\label{fga-lem-4-44}\dcref{4.44}\source{fga-explained:page-0101}\uses{fga-def-4-42}
Every torsor becomes trivial after pulling back along itself, and the two
projections from $X\times_UX$ differ by the unique group element supplied by
the torsor isomorphism.
\end{lemma}
\begin{theorem}\label{fga-thm-4-46}\dcref{4.46}\source{fga-explained:page-0102}\uses{fga-def-4-42,fga-def-4-8,fga-prop-4-43}
If $X\to U$ is a $G$-torsor and $\mathcal X$ is a stack, then restriction gives
an equivalence between $\mathcal X(U)$ and the category of $G$-equivariant
objects of $\mathcal X(X)$.
\end{theorem}
\begin{proof}
Apply stack descent to the covering $X\to U$.  The torsor identity identifies
the overlap with $X\times G$, so a descent isomorphism is exactly an equivariant
structure; cocycle and effectivity are the same conditions.
\end{proof}
