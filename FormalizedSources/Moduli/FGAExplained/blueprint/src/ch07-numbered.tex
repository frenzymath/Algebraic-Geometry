\begin{lemma}[determinant of a finite free resolution]\label{fga-lem-7-1-4}\dcref{7.1.4}\source{fga-explained:page-0172}\uses{}
Let $R$ be a local ring and let $M$ be an $R$-module admitting a finite free
resolution
\[
0\longrightarrow R^r\xrightarrow{\,\varphi\,}R^r\longrightarrow M\longrightarrow0.
\]
The class of $\det(\varphi)$ in $R/R^{\times}$ depends only on $M$, and
$\det(\varphi)$ is a non-zero-divisor of $R$.
\end{lemma}
\begin{proof}
After fixing the quotient $R^r\twoheadrightarrow M$, another presentation with
the same source is obtained by composing $\varphi$ with an automorphism of
$R^r$; its determinant changes by a unit.  Adding a generator to a presentation
and then adding the corresponding relation replaces $\varphi$ by a block matrix
whose determinant is the old determinant times a unit.  Any two finite sets of
generators are related by these elementary additions and removals.  Thus the
class is independent of the resolution.  Since $M$ has finite length at the
generic point of the support, the determinant cannot annihilate a nonzero
element of $R$.
\end{proof}

\begin{remark}[the local multiplicity]\label{fga-rem-7-1-5}\dcref{7.1.5}\source{fga-explained:page-0172}\uses{fga-lem-7-1-4}
Let $X$ be a smooth connected variety over $k$ and let $\mathcal F$ be a
coherent sheaf on $X$ with $\operatorname{Supp}(\mathcal F)\ne X$.  If $V\subset X$
is an irreducible hypersurface with generic point $\eta_V$, put
$R=\mathcal O_{X,\eta_V}$ and $M=\mathcal F_{\eta_V}$.  Then $R$ is a discrete
valuation ring and $M$ has a free resolution
\[
0\longrightarrow R^r\xrightarrow{\,\varphi\,}R^r\longrightarrow M\longrightarrow0.
\]
The two free modules have the same rank.  Define
\[
m_V:=\operatorname{ord}_V(\det\varphi)\in\mathbb Z_{\ge 0}.
\]
Equivalently, on an open neighbourhood on which the resolution is locally free,
$m_V$ is the order of vanishing along $V$ of the determinant of its differential.
By Lemma~\ref{fga-lem-7-1-4}, $m_V$ is independent of the chosen resolution.
\end{remark}

\begin{definition}[divisor of a sheaf]\label{fga-def-7-1-6}\dcref{7.1.6}\source{fga-explained:page-0172}\uses{fga-rem-7-1-5}
Under the assumptions of Remark~\ref{fga-rem-7-1-5}, define
\[
\operatorname{div}(\mathcal F):=\sum_V m_V V,
\]
where the sum runs over irreducible hypersurfaces of $X$.  This is an effective
Cartier divisor; the sum is finite because $m_V\ne0$ only when
$V\subset\operatorname{Supp}(\mathcal F)$.
\end{definition}

\begin{definition}[depth one point]\label{fga-def-7-1-9}\dcref{7.1.9}\source{fga-explained:page-0173}\uses{}
Let $Y$ be a scheme.  A point $p\in Y$ has depth $d$ if
$\operatorname{depth}\mathcal O_{Y,p}=d$.  In particular, $p$ has depth one if
the maximal ideal of $\mathcal O_{Y,p}$ contains a non-zero-divisor and every
element of that maximal ideal becomes a zero-divisor after quotienting by any
non-zero-divisor.
\end{definition}

\begin{remark}[depth one points on a smooth variety]\label{fga-rem-7-1-10}\dcref{7.1.10}\source{fga-explained:page-0173}\uses{fga-def-7-1-9}
If $Y$ is smooth, its depth-one points are precisely the generic points of
prime divisors (closed irreducible hypersurfaces).
\end{remark}

\begin{lemma}[Cartier divisors and codimension one]\label{fga-lem-7-1-11}\dcref{7.1.11}\source{fga-explained:page-0173}\uses{fga-def-7-1-9}
Let $f$ and $g$ be Cartier divisors on a scheme $Y$.  Then $f=g$ if and only if
their stalks agree at every depth-one point of $Y$.
\end{lemma}

\begin{lemma}[determinants in an exact complex]\label{fga-lem-7-1-12}\dcref{7.1.12}\source{fga-explained:page-0174}\uses{}
\textup{(A)} If $\mathcal E$ is a locally free sheaf of rank $r$ on $Y$, then
$\det\mathcal E:=\bigwedge^r\mathcal E$ is an invertible sheaf; a trivialization
is unique up to multiplication by a unit.

\textup{(B)} If
\[
0\longrightarrow\mathcal E_n\longrightarrow\mathcal E_{n-1}\longrightarrow\cdots
\longrightarrow\mathcal E_0\longrightarrow0
\]
is an exact sequence of locally free sheaves, there is a canonical isomorphism
\[
\bigotimes_{i=0}^{n}(\det\mathcal E_i)^{(-1)^i}\;\xrightarrow{\;\sim\;}\;\mathcal O_Y.
\]
\end{lemma}
\begin{proof}
For a short exact sequence $0\to\mathcal E'\to\mathcal E\to\mathcal E''\to0$,
choose locally a basis of $\mathcal E'$ and lifts of a basis of $\mathcal E''$.
Their wedge gives a local isomorphism
$\det\mathcal E'\otimes\det\mathcal E''\simeq\det\mathcal E$,
independent of the lifts.  Iterating this isomorphism proves (B); (A) is the
case of a free sheaf.
\end{proof}

\begin{remark}[incidence correspondence]\label{fga-rem-7-2-2}\dcref{7.2.2}\source{fga-explained:page-0176}\uses{}
For a smooth irreducible surface $S$, the projectivization of the universal ideal
is the blow-up of $S\times S^{[n]}$ along the universal family; the incidence
correspondence $S^{[n,n+1]}$ is the corresponding projective bundle.
\end{remark}

\begin{remark}[obstructions]\label{fga-rem-7-2-4}\dcref{7.2.4}\source{fga-explained:page-0178}\uses{fga-thm-7-2-3}
The proof of smoothness does not assert that the obstruction space
$\operatorname{Ext}^{1}(\mathcal I_Z,\mathcal O_Z)$ vanishes; in general it does
not.
\end{remark}

\begin{remark}[small Hilbert schemes]\label{fga-rem-7-2-5}\dcref{7.2.5}\source{fga-explained:page-0178}\uses{fga-thm-7-2-3}
If $X$ is a nonsingular variety, then $X^{[n]}$ is nonsingular for $n\le3$.
\end{remark}
\begin{proof}
For a cluster $Z$ of length at most three, the tangent space is
$\operatorname{Hom}(\mathcal I_Z,\mathcal O_Z)$.  The local classification of
length-one, length-two, and length-three algebras shows that its dimension is at
most $n\dim X$, with equality on the reduced locus.  The reduced locus is dense,
so the dimension criterion for the Hilbert scheme gives smoothness.
\end{proof}

\begin{remark}[dimension three]\label{fga-rem-7-2-6}\dcref{7.2.6}\source{fga-explained:page-0178}\uses{fga-rem-7-2-5}
If $X$ is nonsingular of dimension $3$, then $X^{[4]}$ is singular.  At the
cluster defined locally by $\mathfrak m^2$ in a regular local ring of dimension
$3$, the Zariski tangent space has dimension strictly larger than $12$.
\end{remark}

\begin{remark}[embedding dimension]\label{fga-rem-7-2-7}\dcref{7.2.7}\source{fga-explained:page-0178}\uses{fga-rem-7-2-5}
Every zero-dimensional subscheme of length at most three has embedding dimension
at most two.  More generally, a cluster of embedding dimension at most two is a
nonsingular point of the Hilbert scheme.
\end{remark}

\begin{remark}[reducibility in higher dimension]\label{fga-rem-7-2-9}\dcref{7.2.9}\source{fga-explained:page-0179}\uses{}
For projective space, $(\mathbb P^d)^{[n]}$ is irreducible for $n<7$ and all
$d$, while $(\mathbb P^d)^{[n]}$ is irreducible for $d=3$ and $n=3$, and is
reducible for $d=3$ and $n>4$.  The same conclusions hold for any smooth
quasi-projective variety of dimension $d$.
\end{remark}

\begin{proposition}[Hilbert scheme of a curve]\label{fga-pro-7-3-3}\dcref{7.3.3}\source{fga-explained:page-0179}\uses{}
Let $C$ be a nonsingular quasi-projective curve.  The Hilbert--Chow morphism
\[
\rho:C^{[n]}\longrightarrow C^{(n)}
\]
is an isomorphism.
\end{proposition}
\begin{proof}
Every local ring of $C$ is a discrete valuation ring, so an ideal of colength
$n$ at $p$ is uniquely $\mathfrak m_p^n$.  Consequently a length-$n$ cluster is
uniquely determined by its effective divisor, and $\rho$ is bijective.  It is
proper and birational, hence an isomorphism by Zariski's main theorem.
\end{proof}

\begin{theorem}[Fogarty]\label{fga-the-7-3-4}\dcref{7.3.4}\source{fga-explained:page-0180}\uses{fga-thm-7-2-3}
If $S$ is a nonsingular quasi-projective surface, then
\[
\rho:S^{[n]}\longrightarrow S^{(n)}
\]
is a resolution of singularities.
\end{theorem}
\begin{proof}
The scheme $S^{[n]}$ is nonsingular and irreducible, and $\rho$ is an
isomorphism over the dense open locus of cycles with distinct support.  This
open locus is the nonsingular locus of $S^{(n)}$; therefore $\rho$ is a proper
birational morphism from a nonsingular scheme and is a resolution.
\end{proof}

\begin{remark}[properties of the resolution]\label{fga-rem-7-3-5}\dcref{7.3.5}\source{fga-explained:page-0180}\uses{fga-the-7-3-4}
The Hilbert scheme of points on a smooth surface gives a canonical resolution of
the singularities of the symmetric power.  The resolution is crepant:
$S^{[n]}$ is Gorenstein and $K_{S^{[n]}}=\rho^*K_{S^{(n)}}$.  If $S$ is a K3
surface or an abelian variety, $S^{[n]}$ is holomorphic symplectic.
\end{remark}

\begin{definition}[punctual Hilbert schemes]\label{fga-def-7-4-3}\dcref{7.4.3}\source{fga-explained:page-0181}\uses{}
For $n\ge1$, let
\[
H_n:=\operatorname{Hilb}^{n}\!\left(\operatorname{Spec}k[[x,y]]\right)
\]
with support at the closed point.  Equivalently, $H_n$ parametrizes ideals of
colength $n$ in $k[[x,y]]$; every such ideal contains $(x,y)^n$.
\end{definition}

\begin{remark}[curvilinear locus]\label{fga-rem-7-4-5}\dcref{7.4.5}\source{fga-explained:page-0182}\uses{fga-def-7-4-3}
The curvilinear locus $H_n^{\mathrm{curv}}\subset H_n$ is irreducible and is a
locally trivial affine bundle of rank $n-2$ over $\mathbb P^1$; in particular,
$\dim H_n^{\mathrm{curv}}=n-1$.
\end{remark}
\begin{proof}
Over the chart of $\mathbb P^1$ where the tangent direction is the $x$-axis, a
curvilinear ideal has the unique form
\[
(y-a_1x-\cdots-a_{n-1}x^{n-1},x^n),
\]
so the chart is affine $(n-1)$-space.  The analogous chart for the other tangent
direction gives the bundle description and irreducibility.
\end{proof}

\begin{proposition}[strict semismallness]\label{fga-pro-7-5-2}\dcref{7.5.2}\source{fga-explained:page-0183}\uses{fga-the-7-3-4,fga-def-7-4-3}
For a smooth projective surface $S$, the Hilbert--Chow morphism
$\rho:S^{[n]}\to S^{(n)}$ is strictly semismall for the stratification by
partitions of $n$, and all its fibers are irreducible.
\end{proposition}
\begin{proof}
For a partition $\nu=(n_1,\ldots,n_r)$, the stratum of $S^{(n)}$ has dimension
$2r$ and codimension $2(n-r)$.  Its fiber is a product
$H_{n_1}\times\cdots\times H_{n_r}$, of dimension $\sum_i(n_i-1)=n-r$;
therefore twice the fiber dimension equals the codimension.  The punctual
Hilbert schemes are irreducible, so the fibers are irreducible.
\end{proof}
