\chapter{The Abel--Jacobi map}
\label{mk-ch-abel-jacobi}

Fix \(P_0\in C(k)\) and set \(J=\operatorname{Pic}^0_{C/k}\).

\begin{definition}[Abel--Jacobi map]
  \label{mk-def-abel-jacobi}
  \dcref{III.6,custom}
  \uses{mk-thm-pic0-abelian}
  \cite{milne-av,mumford-av}
  The line bundle
  \(\mathcal O_{C\times C}(\Delta-C\times P_0-P_0\times C)\), rigidified
  along \(P_0\), defines
  \[
    \iota_{P_0}:C\longrightarrow J,\qquad
    Q\longmapsto[\mathcal O_C(Q-P_0)].
  \]
\end{definition}

\begin{proof}
  The diagonal is an effective Cartier divisor on \(C\times C\).  Subtracting
  the two fibres through \(P_0\) makes its associated line bundle trivial on
  both \(P_0\times C\) and \(C\times P_0\), so it is rigidified.  Restriction
  to \(C\times\{Q\}\) gives \(\mathcal O_C(Q-P_0)\), and the universal
  property of the Picard scheme produces the morphism \(\iota_{P_0}\).
\end{proof}

\begin{theorem}[Symmetric powers]
  \label{mk-thm-symmetric-powers}
  \dcref{III.6,custom}
  \uses{mk-def-abel-jacobi,mk-thm-pic0-abelian}
  For \(n\ge1\), the symmetric power \(C^{(n)}\) is smooth and proper and the
  sum of \(n\) copies of \(\iota_{P_0}\) descends to
  \[
    \alpha_n:C^{(n)}\longrightarrow J,\qquad
    D\longmapsto[\mathcal O_C(D-nP_0)].
  \]
  Its geometric fibres are the complete linear systems of divisors in the
  corresponding class.
\end{theorem}

\begin{proof}
  The Hilbert scheme of length-\(n\) subschemes of a smooth curve is smooth
  and represents effective relative divisors; it is denoted \(C^{(n)}\).
  Locally, a divisor is given by a monic polynomial and the elementary
  symmetric functions provide smooth coordinates.  The tensor product of the
  \(n\) pulled-back rigidified line bundles is invariant under permutations,
  hence descends to \(\alpha_n\).  Two effective divisors have the same image
  exactly when their difference is principal, so the fibre over a line-bundle
  class is its complete linear system.
\end{proof}

\begin{theorem}[Albanese property]
  \label{mk-thm-abel-jacobi-universal}
  \dcref{III.6,custom}
  \uses{mk-def-abel-jacobi,mk-thm-symmetric-powers,mk-thm-rigidity}
  If \(A\) is an abelian variety over \(k\) and \(f:C\to A\) satisfies
  \(f(P_0)=0\), there is a unique homomorphism \(u:J\to A\) such that
  \[
    f=u\circ\iota_{P_0}.
  \]
  Thus \(J\) is the Albanese variety of \((C,P_0)\).
\end{theorem}

\begin{proof}
  For \(n\ge 2g(C)-1\), Riemann--Roch makes
  \(\alpha_n:C^{(n)}\to\operatorname{Pic}^n_{C/k}\) surjective.  Given
  \(f(P_0)=0\), the sum of the values of \(f\) on an ordered \(n\)-tuple is
  symmetric and therefore descends to a morphism
  \(f_n:C^{(n)}\to A\).  On every complete linear system its restriction is
  constant: a morphism from a projective space to an abelian variety is
  constant by the rigidity lemma.  Hence \(f_n\) factors uniquely through
  \(\alpha_n\), giving a morphism \(u_n:\operatorname{Pic}^n_{C/k}\to A\).
  Translating by a degree-\(n\) line bundle identifies this with a morphism
  \(u:J\to A\).  The construction with \(n+1\) and the divisor
  \(P_0\) shows \(u\) is a homomorphism and \(f=u\circ\iota_{P_0}\).
  Surjectivity of \(\alpha_n\) proves uniqueness.
\end{proof}
