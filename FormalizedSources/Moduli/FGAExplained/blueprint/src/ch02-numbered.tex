\begin{remark}[dual Yoneda lemma]\label{fga-rem-2-10}\dcref{2.10}\source{fga-explained:page-0028}\uses{}
For objects $X,Y$ of a category $\mathcal C$, there is a canonical bijection
\[
\operatorname{Hom}_{\mathcal C}(X,Y)
\simeq\operatorname{Nat}\bigl(\operatorname{Hom}_{\mathcal C}(Y,-),
\operatorname{Hom}_{\mathcal C}(X,-)\bigr).
\]
\end{remark}

\begin{remark}[group objects and product-preserving functors]\label{fga-rem-2-14}\dcref{2.14}\source{fga-explained:page-0030}\uses{}
Let $F:\mathcal C\to\mathcal D$ preserve finite products.  If $G$ is a group
object of $\mathcal C$, then $F(G)$ is canonically a group object of $\mathcal D$:
the unit, multiplication, and inverse are obtained by applying $F$ and using
the canonical isomorphisms $F(\mathrm{pt})\simeq\mathrm{pt}$ and
$F(G\times G)\simeq F(G)\times F(G)$.  Thus $F$ induces a functor on group
objects.
\end{remark}

\begin{remark}[discrete group objects]\label{fga-rem-2-23}\dcref{2.23}\source{fga-explained:page-0035}\uses{}
The phrase ``discrete group objects'' refers to group objects obtained from a
category of discrete objects by a finite-product-preserving functor.  The
hypotheses in Definition~2.20 ensure that the characterization of
Proposition~2.22 applies.
\end{remark}

\begin{remark}[pretopologies and topologies]\label{fga-rem-2-25}\dcref{2.25}\source{fga-explained:page-0036}\uses{}
The covering families used to define a Grothendieck topology form a
pretopology.  Equivalent pretopologies induce the same topology and therefore
the same sheaves.  A family of morphisms is jointly surjective when the union
of its set-theoretic images is the whole target.
\end{remark}

\begin{lemma}[separated presheaves]\label{fga-lem-2-43}\dcref{2.43}\source{fga-explained:page-0042}\uses{fga-prop-2-42}
If $F$ is separated for a covering $U=\{U_i\to U\}$, then the restriction map
\[
\operatorname{Nat}(h_U,F)\longrightarrow
\operatorname{Nat}(h_{\!U},F)
\]
is injective, where $h_{\!U}$ is the sieve generated by the covering.
\end{lemma}
\begin{proof}
Two transformations agreeing on the covering agree after restriction to every
$T\to U$, because the pullbacks $T\times_UU_i\to T$ form a covering and the
separatedness condition identifies their sections.
\end{proof}

\begin{proposition}[comparison of topologies]\label{fga-pro-2-49}\dcref{2.49}\source{fga-explained:page-0043}\uses{fga-prop-2-42}
Let $\mathcal T$ and $\mathcal T'$ be Grothendieck topologies on $\mathcal C$.
If $\mathcal T$ is subordinate to $\mathcal T'$, every $\mathcal T'$-sheaf is a
$\mathcal T$-sheaf.  In particular equivalent topologies have the same sheaves.
\end{proposition}
\begin{proof}
The equalizer criterion for sheaves applied to the two classes of covering
families gives the assertion immediately.
\end{proof}

\begin{proposition}[representables on topological spaces]\label{fga-pro-2-54}\dcref{2.54}\source{fga-explained:page-0044}\uses{}
For the category of topological spaces with its classical topology, every
representable functor is a sheaf.  Thus compatible continuous maps from the
members of an open cover glue uniquely.  The analogous statement holds for
representable functors on schemes with the Zariski topology.
\end{proposition}
\begin{proof}
Continuity and equality of maps are local on the source, so compatible maps on
an open cover glue to a unique continuous map.  The scheme case is checked on
affine charts using the sheaf property of regular functions.
\end{proof}

\begin{remark}[the wild flat topology]\label{fga-rem-2-56}\dcref{2.56}\source{fga-explained:page-0045}\uses{}
The topology whose coverings are jointly surjective families of flat morphisms
is strictly finer than the fpqc topology and is not subcanonical.  For an
integral smooth curve $U$ with function field $K$, glue the local schemes
$\operatorname{Spec}\mathcal O_{U,p}$ for closed points $p$ along
$\operatorname{Spec}K$.  The resulting compatible family of maps does not glue
to a morphism from $U$, because the induced set map is not continuous.
\end{remark}

\begin{lemma}[faithfully flat equalizer]\label{fga-lem-2-61}\dcref{2.61}\source{fga-explained:page-0047}\uses{fga-prop-1-13}
If $A\to B$ is faithfully flat, then the sequence
\[
0\longrightarrow A\longrightarrow B\rightrightarrows B\otimes_A B
\]
is exact, where the two arrows are $b\mapsto b\otimes1$ and
$b\mapsto1\otimes b$.
\end{lemma}
\begin{proof}
Injectivity follows from faithful flatness.  After the faithfully flat base
change $A\to B$, the resulting Amitsur complex is split exact: multiplication
$B\otimes_A B\to B$ supplies the contracting extra degeneracy, and the same
homotopy in degree one identifies the equalizer of the two coface maps with
the image of $B$.  Faithful flatness reflects exactness, proving the claim.
\end{proof}
