\chapter{The relative étale Picard functor}
\label{pa-ch-picard}

\begin{definition}[Rigidified line bundles]
  \label{pa-def-rigidified-line-bundle}
  \dcref{2.2}
  \cite{kleiman-picard}
  If \(C_T\to T\) has a section \(s\), a rigidification of \(\mathcal L\) is
  an isomorphism \(s^*\mathcal L\cong\mathcal O_T\).
\end{definition}

\begin{lemma}[Rigidification identifies the quotient]
  \label{pa-lem-rigidification}
  \dcref{2.2}
  \uses{pa-def-rigidified-line-bundle,pa-def-picard}
  \cite{kleiman-picard}
  Relative line-bundle classes modulo \(\operatorname{Pic}(T)\) are
  represented by rigidified bundles, uniquely up to a unique rigidified
  isomorphism.
\end{lemma}

\begin{proof}
  Tensoring by \(p^*s^*\mathcal L^{-1}\), where \(p:C_T\to T\), produces a
  rigidification.  An isomorphism between two such representatives restricts
  to the identity on \(s(T)\), and the quotient ambiguity has been removed.
\end{proof}

\begin{lemma}[The relative Picard presheaf is a group presheaf]
  \label{pa-lem-picard-presheaf-group}
  \dcref{2.2}
  \uses{pa-def-picard}
  \cite{kleiman-picard}
  Pullback makes \(T\mapsto\operatorname{Pic}^{\sharp}_{C/k}(T)\) a
  contravariant commutative group-valued functor.
\end{lemma}

\begin{proof}
  Pullback preserves tensor product, duals, and the trivial bundle, and it
  carries bundles pulled back from \(T\) to bundles pulled back from \(T'\).
  These operations therefore descend to the quotient.
\end{proof}

\begin{lemma}[Etale sheafification preserves the group law]
  \label{pa-lem-etale-sheaf-group}
  \dcref{2.2}
  \uses{pa-lem-picard-presheaf-group}
  \cite{kleiman-picard}
  The etale sheaf associated to
  \(\operatorname{Pic}^{\sharp}_{C/k}\) is a commutative group sheaf.
\end{lemma}

\begin{proof}
  Sheafification is left exact and preserves finite products.  The operations
  and their identities and inverses sheafify from the presheaf identities.
\end{proof}

\begin{proposition}[Invertible Quot locus]
  \label{pa-prop-invertible-quot-locus}
  \dcref{4.8}
  \cite{kleiman-picard,nitsure-hilbert-quot}
  For a fixed Hilbert polynomial, the Quot scheme contains an open locus
  parameterising quotients whose target is an invertible sheaf on \(C_T\).
\end{proposition}

\begin{proof}
  Flatness of the quotient is an open condition, and on a relative smooth
  curve a flat coherent quotient with rank one and fixed Hilbert polynomial is
  locally free of rank one.  The invertibility locus is therefore open.
\end{proof}

\begin{lemma}[Generator changes give the Picard equivalence]
  \label{pa-lem-generator-equivalence}
  \dcref{4.8}
  \uses{pa-prop-invertible-quot-locus}
  \cite{kleiman-picard}
  On the invertible Quot locus, two evaluation quotients define the same
  Picard class exactly when they differ fppf-locally by a change of basis of
  the generating module.
\end{lemma}

\begin{proof}
  An isomorphism of target line bundles carries one chosen generating basis to
  the other.  Conversely, a basis change gives an isomorphic quotient.  The
  matrices are defined after the fppf cover on which the generating module is
  free.
\end{proof}

\begin{lemma}[Universal class after etale descent]
  \label{pa-lem-universal-class-descent}
  \dcref{4.3}
  \uses{pa-lem-rigidification,pa-lem-generator-equivalence}
  \cite{kleiman-picard}
  Local universal line bundles on the Quot charts differ on overlaps by
  pullbacks from the base, and hence define a universal class in the etale
  Picard sheaf.
\end{lemma}

\begin{proof}
  The ratio of two line bundles representing the same relative class is
  pulled back from the base by the quotient relation.  These ratios satisfy
  the cocycle condition, so etale descent glues the classes; changing a
  representative tensors it by a base line bundle.
\end{proof}

\begin{lemma}[Flattening stratification for Quot]
  \label{pa-lem-flattening-stratification}
  \dcref{1.1}
  \cite{nitsure-hilbert-quot}
  A coherent Quot family has a universal locally closed stratification of the
  base on which it is flat with fixed Hilbert polynomial.
\end{lemma}

\begin{proof}
  After a projective embedding, Fitting ideals of finitely many graded pieces
  cut out the prescribed ranks.  Their universal property gives the required
  stratification.
\end{proof}

\begin{lemma}[The invertible locus is open]
  \label{pa-lem-invertible-open}
  \dcref{4.8}
  \uses{pa-lem-flattening-stratification,pa-prop-invertible-quot-locus}
  \cite{kleiman-picard}
  The locus of invertible rank-one quotients in a flat Quot family is open.
\end{lemma}

\begin{proof}
  The locally free locus of a flat coherent sheaf is open; fixed rank one
  identifies it with the invertible quotient locus.
\end{proof}

\begin{definition}[Relative Picard functor]
  \label{pa-def-picard}
  \dcref{2.2}
  \cite[Def.~2.2]{kleiman-picard}
  For \(T\in(\Sch/k)\), put \(C_T=C\times_kT\) and
  \[
    \operatorname{Pic}^{\sharp}_{C/k}(T)=
    \operatorname{Pic}(C_T)/\operatorname{Pic}(T).
  \]
  Let \(\operatorname{Pic}_{(C/k)\et}\) be its associated etale sheaf.
\end{definition}

\begin{proposition}[Group law]
  \label{pa-prop-picard-group}
  \dcref{2.2}
  \uses{pa-def-picard}
  \cite{kleiman-picard}
  Tensor product, the trivial bundle, and duality make
  \(\operatorname{Pic}_{(C/k)\et}\) a commutative group-valued functor.
\end{proposition}

\begin{proof}
  Tensor product, the trivial bundle, and dualisation define the group
  operations on \(\operatorname{Pic}(C_T)\).  They are compatible with
  pullback from \(T\), and hence descend to the quotient and its etale
  sheaf.  The canonical symmetry and associativity constraints make the
  resulting group commutative.
\end{proof}

\begin{theorem}[Etale Picard representability]
  \label{pa-thm-picard-representability}
  \dcref{4.8}
  \uses{pa-def-picard}
  \cite[Thm.~4.8]{kleiman-picard}
  The sheaf \(\operatorname{Pic}_{(C/k)\et}\) is represented by a separated
  \(k\)-scheme \(\operatorname{Pic}_{C/k}\), locally of finite type, with open
  quasi-projective degree pieces
  \[
    \operatorname{Pic}_{C/k}=\coprod_{d\in\mathbb Z}\operatorname{Pic}^d_{C/k}.
  \]
\end{theorem}

\begin{proof}
  Choose a relatively ample line bundle on \(C\).  For \(n\) sufficiently
  large, every line bundle in a fixed degree range is \(n\)-regular and is
  represented by a quotient of a fixed finite-dimensional space of sections.
  Grothendieck's Quot scheme supplies the parameter space.  The change of
  generators gives an etale-locally trivial equivalence relation, and
  faithfully flat descent identifies its quotient with the corresponding
  open of the Picard sheaf.  Gluing degree ranges yields the separated,
  locally finite-type scheme and its open quasi-projective components.
\end{proof}

\begin{proposition}[Universal line bundle]
  \label{pa-prop-poincare}
  \dcref{4.3}
  \uses{pa-thm-picard-representability}
  \cite[Ex.~4.3]{kleiman-picard}
  After etale localisation on \(\operatorname{Pic}_{C/k}\), the universal class
  is represented by a line bundle on
  \(C\times_k\operatorname{Pic}_{C/k}\), unique up to tensoring by a pullback
  from \(\operatorname{Pic}_{C/k}\).
\end{proposition}

\begin{proof}
  The universal quotient on the Quot scheme gives a universal line bundle
  after restricting to the invertible locus.  On overlaps two choices differ
  by tensoring with the pullback of a line bundle from the parameter space.
  The etale topology kills this descent obstruction, producing a universal
  class on \(C\times_k\operatorname{Pic}_{C/k}\), unique up to such a
  pullback.
\end{proof}
