\section{Direct limits}

\begin{theorem}[Tensor products commute with direct limits]
\label{mcrt-thm-a-1}\dcref{A.1}
\source{matsumura-commutative-ring-theory:page-0285}\uses{}
For a directed system $(M_i)_{i\in I}$ of $A$-modules and an $A$-module $N$,
there is a natural isomorphism
\[
\left(\varinjlim_{i\in I}M_i\right)\otimes_A N
\;\cong\;\varinjlim_{i\in I}(M_i\otimes_A N).
\]
\end{theorem}
\begin{proof}
The maps $M_i\otimes_A N\to(\varinjlim M_i)\otimes_A N$ induce a map in the
display.  Conversely, represent an elementary tensor by $m_i\otimes n$ and
send it to the class of that tensor in the direct limit.  Directedness makes
this independent of representatives, and the two maps are inverse on
elementary tensors, hence on the tensor products.
\end{proof}

\begin{theorem}[Exactness of direct limits]
\label{mcrt-thm-a-2}\dcref{A.2}
\source{matsumura-commutative-ring-theory:page-0286}\uses{}
Let
\[
0\longrightarrow(M'_i)_{i\in I}\xrightarrow{\,\alpha_i\,}
(M_i)_{i\in I}\xrightarrow{\,\beta_i\,}(M''_i)_{i\in I}
\longrightarrow0
\]
be a morphism of directed systems of $A$-modules, exact at every index.
Then
\[
0\longrightarrow\varinjlim_i M'_i\longrightarrow\varinjlim_i M_i
\longrightarrow\varinjlim_i M''_i\longrightarrow0
\]
is exact.  Thus direct limits are exact functors on $A$-modules.
\end{theorem}
\begin{proof}
The first map is injective because an element represented by $x_i\in M'_i$
maps to zero only after passage to some $j\geq i$, where exactness gives
$\alpha_j(x_j)=0$ and hence $x_j=0$.  For surjectivity, represent an element
of $\varinjlim M''_i$ by $y_i$ and lift it to $x_i\in M_i$ using surjectivity
of $\beta_i$.  If the image of $x_i$ is zero in the limit, then after increasing
the index its image is zero, so exactness gives a preimage in the corresponding
$M'_j$.  These two observations prove exactness in the middle and at the ends.
\end{proof}
