Fix \(P_0\in C(k)\) and set \(J=\operatorname{Pic}^0_{C/k}\).

\begin{lemma}[The diagonal is an effective Cartier divisor]
  \label{mk-lem-diagonal-cartier}
  \dcref{III.6,custom}
  \cite{hartshorne-ag,milne-av}
  The diagonal \(\Delta\subset C\times_kC\) is an effective Cartier divisor.
\end{lemma}

\begin{proof}
  The diagonal is a regular immersion of codimension one because \(C/k\) is
  smooth of relative dimension one.  A regular codimension-one immersion is
  effective Cartier.
\end{proof}

\begin{lemma}[The pointed diagonal bundle is rigidified]
  \label{mk-lem-pointed-diagonal-rigidified}
  \dcref{III.6,custom}
  \uses{mk-lem-diagonal-cartier}
  \cite{milne-av}
  The bundle
  \(\mathcal P=\mathcal O(\Delta-C\times P_0-P_0\times C)\) restricts
  trivially to both \(C\times\{P_0\}\) and \(\{P_0\}\times C\).
\end{lemma}

\begin{proof}
  Restriction of \(\Delta\) to either axis is the point \(P_0\), while the
  corresponding fibre contributes the same divisor and the other fibre is
  disjoint.  The two restrictions are therefore \(\mathcal O_C\), with their
  canonical trivializations.
\end{proof}

\begin{lemma}[The ordered Abel map]
  \label{mk-lem-ordered-abel-map}
  \dcref{III.6,custom}
  \uses{mk-def-abel-jacobi}
  \cite{milne-av}
  The map \(C^n\to J\), \((Q_1,\dots,Q_n)\mapsto
  \sum_i[\mathcal O_C(Q_i-P_0)]\), is a morphism.
\end{lemma}

\begin{proof}
  It is the \(n\)-fold composite of \(\iota_{P_0}\) with the group law on
  \(J\), hence is a morphism.
\end{proof}

\begin{lemma}[Symmetric descent of the ordered Abel map]
  \label{mk-lem-symmetric-abel-descent}
  \dcref{III.6,custom}
  \uses{mk-lem-ordered-abel-map}
  \cite{milne-av}
  The ordered Abel map is invariant under \(S_n\) and descends uniquely to
  \(C^{(n)}\).
\end{lemma}

\begin{proof}
  Permuting the \(Q_i\) permutes tensor factors of the associated line bundle;
  the symmetry constraint gives equality in the commutative Picard group.
  The quotient property of \(C^{(n)}\) gives the descent.
\end{proof}

\begin{lemma}[Fibres of the Abel map are complete linear systems]
  \label{mk-lem-abel-fibre-linear-system}
  \dcref{III.6,custom}
  \uses{mk-def-divisor,mk-def-divisor-line-bundle}
  \cite{milne-av,hartshorne-ag}
  Two effective divisors \(D,E\) of the same degree have the same image in
  \(\operatorname{Pic}^n_{C/k}\) exactly when \(D-E\) is principal; hence a
  geometric fibre is \(|D|\cong\PP H^0(C,\mathcal O_C(D))\).
\end{lemma}

\begin{proof}
  Equality of the line-bundle classes gives a rational function
  \(f\) with \(E=D+\operatorname{div}(f)\), and the converse is immediate.
  Effective divisors linearly equivalent to \(D\) are the zero divisors of
  nonzero sections of \(\mathcal O_C(D)\), modulo scalar multiplication.
\end{proof}

\begin{lemma}[Large-degree Abel map is surjective]
  \label{mk-lem-abel-jacobi-surjective}
  \dcref{III.6,custom}
  \uses{mk-cor-large-degree,mk-lem-abel-fibre-linear-system}
  \cite{milne-av}
  For \(n\ge2g(C)-1\), every degree-\(n\) line bundle is represented by an
  effective divisor.
\end{lemma}

\begin{proof}
  Riemann--Roch and large-degree vanishing give
  \(h^0(C,\mathcal L)=n+1-g(C)>0\) for every \(\deg\mathcal L=n\).  The zero
  divisor of a nonzero section represents \(\mathcal L\).
\end{proof}

\begin{lemma}[Constancy on a linear system]
  \label{mk-lem-map-constant-on-linear-system}
  \dcref{III.6,custom}
  \uses{mk-lem-abel-fibre-linear-system,mk-thm-rigidity}
  \cite{milne-av}
  A morphism from a complete linear system \(|D|\cong\PP^r\) to an abelian
  variety is constant.
\end{lemma}

\begin{proof}
  A projective space is proper and connected.  Apply the rigidity lemma to
  the morphism \(\PP^r\times\Spec k\to A\); its image is a complete connected
  subscheme of an affine neighbourhood of \(A\), hence a point.
\end{proof}

\begin{definition}[Abel--Jacobi map]
  \label{mk-def-abel-jacobi}
  \dcref{III.6,custom}
  \uses{mk-thm-pic0-abelian}
  \cite{milne-av,mumford-av}
  The line bundle
  \(\mathcal O_{C\times C}(\Delta-C\times P_0-P_0\times C)\), rigidified
  along \(P_0\), defines
  \[
    \iota_{P_0}:C\longrightarrow J,\qquad
    Q\longmapsto[\mathcal O_C(Q-P_0)].
  \]
\end{definition}

\begin{proof}
  The diagonal is an effective Cartier divisor on \(C\times C\).  Subtracting
  the two fibres through \(P_0\) makes its associated line bundle trivial on
  both \(P_0\times C\) and \(C\times P_0\), so it is rigidified.  Restriction
  to \(C\times\{Q\}\) gives \(\mathcal O_C(Q-P_0)\), and the universal
  property of the Picard scheme produces the morphism \(\iota_{P_0}\).
\end{proof}

\begin{theorem}[Symmetric powers]
  \label{mk-thm-symmetric-powers}
  \dcref{III.6,custom}
  \uses{mk-def-abel-jacobi,mk-thm-pic0-abelian}
  For \(n\ge1\), the symmetric power \(C^{(n)}\) is smooth and proper and the
  sum of \(n\) copies of \(\iota_{P_0}\) descends to
  \[
    \alpha_n:C^{(n)}\longrightarrow J,\qquad
    D\longmapsto[\mathcal O_C(D-nP_0)].
  \]
  Its geometric fibres are the complete linear systems of divisors in the
  corresponding class.
\end{theorem}

\begin{proof}
  The Hilbert scheme of length-\(n\) subschemes of a smooth curve is smooth
  and represents effective relative divisors; it is denoted \(C^{(n)}\).
  Locally, a divisor is given by a monic polynomial and the elementary
  symmetric functions provide smooth coordinates.  The tensor product of the
  \(n\) pulled-back rigidified line bundles is invariant under permutations,
  hence descends to \(\alpha_n\).  Two effective divisors have the same image
  exactly when their difference is principal, so the fibre over a line-bundle
  class is its complete linear system.
\end{proof}

\begin{theorem}[Albanese property]
  \label{mk-thm-abel-jacobi-universal}
  \dcref{III.6,custom}
  \uses{mk-def-abel-jacobi,mk-thm-symmetric-powers,mk-thm-rigidity}
  If \(A\) is an abelian variety over \(k\) and \(f:C\to A\) satisfies
  \(f(P_0)=0\), there is a unique homomorphism \(u:J\to A\) such that
  \[
    f=u\circ\iota_{P_0}.
  \]
  Thus \(J\) is the Albanese variety of \((C,P_0)\).
\end{theorem}

\begin{proof}
  For \(n\ge 2g(C)-1\), Riemann--Roch makes
  \(\alpha_n:C^{(n)}\to\operatorname{Pic}^n_{C/k}\) surjective.  Given
  \(f(P_0)=0\), the sum of the values of \(f\) on an ordered \(n\)-tuple is
  symmetric and therefore descends to a morphism
  \(f_n:C^{(n)}\to A\).  On every complete linear system its restriction is
  constant: a morphism from a projective space to an abelian variety is
  constant by the rigidity lemma.  Hence \(f_n\) factors uniquely through
  \(\alpha_n\), giving a morphism \(u_n:\operatorname{Pic}^n_{C/k}\to A\).
  Translating by a degree-\(n\) line bundle identifies this with a morphism
  \(u:J\to A\).  The construction with \(n+1\) and the divisor
  \(P_0\) shows \(u\) is a homomorphism and \(f=u\circ\iota_{P_0}\).
  Surjectivity of \(\alpha_n\) proves uniqueness.
\end{proof}
